أَلَمْ تَرَ أَنَّ اللَّهَ أَنزَلَ مِنَ السَّماءِ مَاءً فَأَخْرَجْنَا بِهِ ثَمَرَاتٍ مُخْتَلِفاً أَلْوَانُهَا وَمِنَ الْجِبَالِ جُدَدٌ بِيضٌ وَحُمْرٌ مُخْتَلِفٌ أَلْوَانُهَا وَغَرَابِيبُ سُودٌ ٭ وَمِنَ النَّاسِ وَالدَّوَابِّ وَالْأَنْعَامِ مُخْتَلِفٌ أَلْوَانُهُ كَذلِكَ
﴾ مردمهايي هستند كه رنگهايشان فرق مي‌كند، دامهايي هستند كه رنگهايشان فرق مي‌كند، سنگهايي هستند كه رنگهايشان فرق مي‌كند همين، اينها را كه ديگر در رديف ملائكه ذكر نمي‌كند شما در بحث علم باب
بحارالأنوار
مي‌بينيد وجود مبارك امام سجاد(سلام الله عليه) مي‌فرمايد انسان وقتي در فضاي علم قدم گذاشت خداي سبحان نامِ علما را كنار ملائكه ذكر مي‌كند
[36]
اين در
بحارالأنوار
در بحث فضيلت علم هست كه ﴿
شَهِدَ اللّهُ أَنَّهُ لاَ إِلهَ إِلَّا هُوَ وَالْمَلاَئِكَةُ وَأُولُوا الْعِلْمِ
﴾
[37]
اين بقيه كه گفته شد ديگر در آن روايات نيست به استناد آ‌ن تحليل روايي مي‌شود از اين آيات هم اين مطالب را گرفت كه گاهي مجاورت، پيامدار است آنجا مجاورت علما با ملائكه پيام خاص داشت، اين آيات يادشده هم بي‌پيام نيست فرمود قدري خودتان بخوريد، قدري به دامهايتان بدهيد اين همين است ديگر اگر فرمود: ﴿
مَتَاعاً لَّكُمْ وَلْأَنْعَامِكُمْ
﴾
[38]
، اگر فرمود: ﴿
كُلُوا وَارْعَوْا أَنْعَامَكُمْ
﴾
[39]
اگر فرمود: ﴿
وَمِنَ النَّاسِ وَالدَّوَابِّ وَالْأَنْعَامِ مُخْتَلِفٌ أَلْوَانُهُ
﴾ همين است پس اگر آن كرامت را، آن گوهر را كسي حفظ كرد مشمول آيه سورهٴ مباركهٴ «آل‌عمران» خواهد شد، در رديف ملائكه قرار مي‌گيرد و ﴿
وَالْمَلاَئِكَةُ وَأُولُوا الْعِلْمِ
﴾ مي‌شود.
در سالهاي قبل هم اين بيان لطيف از بزرگان نقل شده است اينكه مرحوم كليني نقل كرده، در مقدمه
معالِم
هم آمده در خيلي از جوامع روايي ما هست كه ملائكه پَرهاي خود را پهن مي‌كنند زير پاي طالبان علوم «
إنّ الملائكة لتضع أجنحتها لطلاّب العلم
»
[40]
اين روايت معنا شده است فرشته‌ها پَر پهن مي‌كنند، پَر پهن مي‌كنند يعني چه؟ يعني شما هم مثل ما پَر دربياوريد، مثل ما پَر دربياوريد كه ﴿
لِيُنذِرُوا قَوْمَهُمْ إِذَا رَجَعُوا إِلَيْهِمْ
﴾
[41]
برويد دنبال پروانهٴ كسب و نان و آب؟ خب آن پَر، پَر همين مرغهايي است كه به دنبال تالاب حركت مي‌كنند فرشته با پَرش چه كار مي‌كند؟ اين زمستان و تابستان مهاجرت مي‌كند به طمع طمعهٴ تالاب حركت مي‌كند يا فرشته‌ها حركت مي‌كنند از جهت به بي‌جهتي، از طبيعت به فراطبيعت اينها زير بال پاهاي ما پَر پهن مي‌كنند كه به ما پَر درآوردن، پرواز كردن و جهت‌يابي ياد بدهند پَر دربياور، پَر بكش، از طبيعت برو بيرون وگرنه لازم نيست كه به مَلك پخش كنند ديگر، اذان است ديگر.

اعوذ بالله من الشيطان الرجيم
بسم الله الرحمن الرحيم
﴿
وَلَقَدْ كَرَّمْنَا بَنِي آدَمَ وَحَمَلْنَاهُمْ فِي الْبَرِّ وَالْبَحْرِ وَرَزَقْنَاهُم مِنَ الطَّيِّبَاتِ وَفَضَّلْنَاهُمْ عَلَي كَثِيرٍ مِمَّنْ خَلَقْنَا تَفْضِيلاً﴿70
﴾
بعد از اينكه برخي از نعمتهاي دريايي و صحرايي را برشمردند و اينكه انسان به اذن خدا مسخّر بر اين امور است و اگر خطري پيش آمد تنها راه نجات از خطر توجّه به ذات اقدس الهي است جريان آفرينش انسان را ذكر فرمود، فرمود ما انسان را گرامي داشتيم كرامت همان طوري كه ملاحظه فرموديد يك معناي ارزشي را به همراه دارد بر خلاف كبير، عظيم، كثير، اينها گاهي بر جماد، گاهي بر نبات، گاهي بر حيوان اطلاق مي‌شود، مي‌شود گفت كه سنگِ بزرگ، كوهِ بزرگ، درخت بزرگ اما سنگ كريم يا درخت كريم اينها كم گفته مي‌شود مگر با قرينه همراه باشد.
كرامت غير از كبير بودن، عظيم بودن و مانند آن، آن معناي ارزشي و معنوي را به همراه دارد هم جود و سخا را در درون خود نهادينه كرده دارد، هم عفو و بخشش و گذشت را دارد و هم معنويّتهاي ديگر را. از اينكه ذات اقدس الهي عرش را به كرامت وصف كرد كه عرشْ كريم است ﴿
رَبُّ الْعَرْشِ الْكَرِيمِ
﴾
[1]
از اينكه قرآن را به كرامت ستود ﴿
إِنَّهُ لَقُرْآنٌ كَرِيمٌ
﴾
[2]
از اينكه شعيب(سلام الله عليه) نبوّت را رزق كريم دانست ﴿
وَأَعْتَدْنَا لَهَا رِزْقاً كَرِيماً
﴾
[3]
و از اينكه ذات اقدس الهي ملائكه را به عنوان عبادِ مُكرَم معرّفي كرد در سورهٴ مباركهٴ «انبياء» فرمود هرگز فرشته‌ها ادّعاي ربوبيّتي ندارند ﴿
بَلْ عِبَادٌ مُّكْرَمُونَ ٭ لاَ يَسْبِقُونَهُ بِالْقَوْلِ وَهُم بِأَمْرِهِ يَعْمَلُونَ
﴾
[4]
ملائكه بندگان مكرّم الهي‌اند چه اينكه در آيات ديگر فرشتگاني كه در مسير وحي الهي‌اند آنها را هم به عنوان بندگان كريم ياد كرده است ﴿
بِأَيْدِي سَفَرَةٍ ٭ كِرَامٍ بَرَرَةٍ
﴾
[5]
فرشتگاني هم كه مأمور ضبط اعمال ما و ثبت احوال ما هستند آنها را به كِرام وصف كرد فرمود: ﴿
إِنَّ عَلَيْكُمْ لَحَافِظِينَ ٭ كِرَاماً كَاتِبِينَ
﴾
[6]
، اگر ذات اقدس الهي عرش را، قرآن را، فرشته‌ها را به كَرم ستود معلوم مي‌شود يك معناي ارزشي را به همراه دارد صِرف بزرگي و بلندي و امثال ذلك كه بر احجام و اجسام هم اطلاق مي‌شود نيست.
مطلب ديگر اين بود كه اين كرامت كه گوهر گرانمايه است و به اولياي الهي اسناد داده مي‌شود مخصوصاً دربارهٴ بعضي از ائمه(عليهم السلام) مثل امام مجتبي(سلام الله عليه) گفته مي‌شود كه كريمِ اهل‌بيت است يا دربارهٴ فاطمه معصوم(سلام الله عليهما) گفته مي‌شود كريمهٴ اهل‌بيت است يا دربارهٴ وجود مبارك ثامن‌الحجج كه او كريم است، امامِ كريم است در اين‌گونه از موارد معناي سخا و جود و بخشش و عفو و گذشت و شفاعت و اينها تعبيه شده است قهراً يك واژهٴ عربيِ خاصّ است كه اگر بخواهيم به فارسي ترجمه كنيم ناچاريم از چند كلمه كمك بگيريم بعضي از لغتهاي عربي معادل فارسي به اين معنا ندارند كه يك كلمهٴ عربي را به يك كلمهٴ فارسي ما ترجمه بكنيم بلكه ناچاريم چند كلمه را مركّب بكنيم از تركيب چند كلمه بتوانيم ترجمهٴ آن كلمهٴ بسيط را حفظ كرده باشيم.
مستحضريد كه اگر يك كلمهٴ بسيطي يك معناي جامعي داشت ما خواستيم به وسيلهٴ چند كلمه و چند مفهوم آن معناي جامع را ارائه كنيم بالأخره موفق نمي‌شويم ريزشي هم خواهد داشت، چرا؟ براي اينكه آن معنا بسيط است و كلمهٴ واحد آن معناي بسيط را در لغت عرب تأمين مي‌كند و مي‌فهماند ولي ما اگر خواستيم به فارسي ترجمه كنيم چون ناچاريم از چند كلمه كمك بگيريم اين كلمات همان طوري كه از نظر لفظ از هم بُريده و جداي‌اند مفاهيم اينها هم از هم گسيخته است وقتي ما با پنج مفهوم يا شش مفهوم خواستيم يك معناي بسيطي را تفهيم كنيم اين معاني و مفاهيم متعدّد هر كدام به اندازهٴ خود ظرفيّتي دارند كه آن مفهوم بسيط را گوشه‌اي از آن را نشان مي‌دهند با اينكه او بسيط است و چون اين مفاهيم از هم جداي‌اند مثل اين بَندهاي غربال است كه هر كدام يك نَم خاصّي را مي‌گيرند اين وسطها خالي است آن مفهوم مي‌ريزد، بقيه مطالب مي‌ريزند اين است كه مي‌گويند در بعضي از امور «يُدرك ولا يوصف» چرا «يُدرك ولا يوصف» براي اينكه آنچه را كه انسان مي‌يابد يك معناي جامع و بسيط است كلمه‌اي ندارد كه آن معنا را تفهيم كند و منتقل كند ناچار از چند كلمه كمك مي‌گيرد هر كدام از اين كلمه مثل بَند غربال‌اند كه به اندازهٴ خود نَم يا آب را حمل مي‌كنند چون اين كلمات از هم جداي‌اند يك، مفاهيم اينها از هم جداي‌اند دو، آن معناي بسيط كه بخواهد تقسيم بشود در اين بندهاي غربال ريزشي هم دارد سه، لذا گاهي انسان يك معناي بسيط را مي‌يابد ولي بخواهد تفهيم كند مقدورش نيست چهار، اگر ما معادل فارسي مي‌داشتيم با همان يك كلمه آن معنا را تفهيم مي‌كرديم ولي چون معادل فارسي نيست قهراً از چند كلمه بايد كمك بگيريم همين مشكل را به همراه دارد كرامت ظاهراً از همين قبيل است يعني واژه‌اي نيست كه ما به وسيله كبير، عظيم، سَخي، جواد و مانند آن بتوانيم او را تفهيم كنيم بلكه عصاره همه اين مفاهيم در كرامت تأمين شده است، اين هم يك مطلب.
فرشته‌ها كه كريم‌اند منشأ كرامت آنها علم است و تقوا، عرش كه كريم است براي اينكه مركز علم است، قرآن كه كريم است براي اينكه علم است و نزاهت از تحريف و مانند آن را به همراه دارد اين مجموعه نشان مي‌دهد كه گوهري كه از نقص از يك سو، از عيب از سوي ديگر منزّه باشد اين گوهر را مي‌گويند كريم اين هم يك مطلب.
دربارهٴ انسان فرمود من او را با گوهرِ كريم آفريدم چه مؤمن، چه كافر انسان بي‌نقص خلق شده است روحي كه ذات اقدس الهي به انسان داد اين روح شرافت اضافهٴ به الله را به همراه دارد كه ﴿
نَفَخْتُ فِيهِ مِن رُّوحِي
﴾
[7]
حالا بعد عده‌اي به سوء اختيار خود ﴿
قَدْ خَابَ مَن دَسَّاهَا
﴾
[8]
اين گوهر كريم را دفن مي‌كنند گاهي هم به حُسن اختيار خود اين گوهر كريم را شكوفا مي‌كنند آن كرامتهاي كسبي با تقوا و مانند آن حاصل مي‌شود كه ﴿
إِنَّ أَكْرَمَكُمْ عِندَ اللَّهِ أَتْقَاكُمْ
﴾
[9]
اما اين كرامت نفسي و گوهر دروني اين عطيهٴ الهي است ذات اقدس الهي اين گوهر را آفريد و براي خلافت هم خلق كرد كه بحث خلافت در ديروز گذشت، اگر جريان كرامت به اين گوهر برمي‌گردد كه اين بايد خليفةالله باشد طبق تحليل بحث ديروز خليفه آن است كه حرف مستخلَف‌عنه را بزند، پيام مستخلف‌عنه را ابلاغ كند و برابر فرمان مستخلف‌عنه عمل كند وگرنه اين حيات او مي‌شود غصب، اين سِمتهاي او مي‌شود غصب، آن‌گاه استحقاق ﴿
أُولئِكَ كَالْأَنْعَامِ بَلْ هُمْ أَضَلُّ
﴾
[10]
را پيدا مي‌كند يا ﴿
شَيَاطِينَ الْإِنْسِ وَالْجِنِّ
﴾
[11]
شدن را پيدا مي‌كند و مانند آن.

ذات اقدس الهي هم مشخص كرده است كه من چگونه انسان را گرامي داشتم. مستحضريد كه جناب فخررازي ده وجه ذكر كرده براي گراميداشت عقل و وجوه فراواني از خودش و از ديگران نقل كرده كه چرا انسان كريم است. در بحثهاي دو، سه روز قبل اشاره شد كه آن ساده‌ترين حرف همين حرفهاي ابتدايي است كه گاهي در ادبيات نثر، گاهي در ادبيات نظم آمده كه انسان كريم است براي اينكه در دوران كودكي شيري كه مادرش به او مي‌دهد از بالاي سينه به او شير مي‌دهد ولي حيوانات مثل گوسفند و امثال گوسفند وقتي شير مي‌خورند به تعبير همين شعرا از «نيمِ زيرينه بُوَد» شيرِ انسان از سرِ سينه بوَد شير حيوانات و اينها، خب اينها تعبيرات عادي است اينها باعث كرامت نيست انسان مستوي‌القامه است آنها منحني‌القامه‌اند آنها سرشان را خم مي‌كنند دهنشان را پيش غذا مي‌برند، انسان غذا را پيش دهن مي‌آورد اينها بد نيست اما اينها معيار كرامت نيست. انسان ولو در حدّ بلال حبشي(رضوان الله عليه) باشد كريم است، از آن طرف طاوس و طيهو با آن زيبايي‌شان مثلاً اين سِمت را ندارند يا آن روميهاي زيبا اين سِمت را ندارند اينها جزء فروعات مسئله است. عمده آن است كه ما ببينيم قرآن كريم منشأ كرامت انسان را چه چيزي مي‌داند نه كرامتهاي كسبي كه فرمود: ﴿
إِنَّ أَكْرَمَكُمْ عِندَ اللَّهِ أَتْقَاكُمْ

وَنَفَخْتُ فِيهِ مِن رُّوحِي

خَلَقْتُ بِيَدَيَّ
﴾

﴾

﴾
[27]
وگرنه اگر اين نقش نداشت ممكن بود بفرمايد «إقرأ وربك الأكرم الذي خلق السماوات السبع والأرضين وما فيهنّ» خلقت آسمان و زمين ما را به اكرم بودن خدا هدايت نمي‌كند، خلقت كريم ما را به اكرم‌بودن الله منتقل مي‌كند اگر خواستيم برهان «لِمّ» اقامه كنيم چون خدا كريم است از كريم يك موجود باكرامت صادر مي‌شود، اگر خواستيم از برهان «إنّ» كمك بگيريم چون انسان كريم است منشأ پيدايش انسان كريم خداي اكرم خواهد بود اينها عصارهٴ كرامتهاي نفسي انسان است كه انسان با آن علمِ الهي كه به ﴿
عَلَّمَ الْإِنسَانَ مَا لَمْ يَعْلَمْ
﴾
[28]
بر اساس ﴿
فَأَلْهَمَهَا فُجُورَهَا وَتَقْوَاهَا
﴾
[29]
اين مي‌شود كرامت نفسي، كرامتهاي كسبي هم با علمِ اضافي، با تقواي اضافي كه اينها اضافه بر گوهر ذات است اينها منشأ پيدايش كرامت‌اند و ذات اقدس الهي عده‌اي را هم غير از ملائكه و غير از انبيا و اوليا به عنوان كريم ستوده است.
مستحضريد شهيد را قرآن كريم، كريم مي‌داند در سورهٴ مباركهٴ «يس» وقتي پيام شهيد را نقل مي‌كند مي‌فرمايد آيه 26 و 27 سورهٴ مباركهٴ «يس» اين است ﴿
قِيلَ ادْخُلِ الْجَنَّةَ قَالَ يَا لَيْتَ قَوْمِي يَعْلَمُونَ ٭ بِمَا غَفَرَ لِي رَبِّي وَجَعَلَنِي مِنَ الْمُكْرَمِينَ
﴾ اين شهيد سورهٴ «يس» است وقتي اين را شهيد كردند اين وارد بهشت شد بهشت برزخي همان طوري كه ما در زيارت شهدا به اينها عرض مي‌كنيم «
يا ليتني كنتُ معكم
» در وارث و غير وارث «
يا ليتني كنتُ معكم فأفوز معكم
» اي كاش ما با شما شهدا بوديم و فوز عظيم نصيب ما مي‌شد شهدا هم مي‌گويند اي كاش با ما بوديد مي‌ديد خدا با ما چه كرد اين ﴿
يَا لَيْتَ
﴾ حرف دو طرف است زائر به شهيد مي‌گويد «
يا ليتني كنّتُ معكم فأفوز معكم
» شهيد مي‌گويد: ﴿
يَا لَيْتَ قَوْمِي يَعْلَمُونَ ٭ بِمَا غَفَرَ لِي رَبِّي وَجَعَلَنِي مِنَ الْمُكْرَمِينَ
﴾ اي كاش بازمانده‌ها، قبيلهٴ من، خانوادهٴ من اينها مي‌دانستند كه خدا با ما چه كرد، با ما چه كرد؟ ما را از مُكرَمين قرار داد، خب مكرمين در فرهنگ قرآن چه گروهي‌اند؟ در سورهٴ مباركهٴ «انبياء» داشتيم ﴿
بَلْ عِبَادٌ مُّكْرَمُونَ ٭ لاَ يَسْبِقُونَهُ بِالْقَوْلِ وَهُم بِأَمْرِهِ يَعْمَلُونَ
﴾
[30]
در ساير سوَر هم داشتيم ﴿
إِنَّ عَلَيْكُمْ لَحَافِظِينَ ٭ كِرَاماً كَاتِبِينَ
﴾
[31]
﴿ ﴾
[32]
و مانند آن ﴿
بِأَيْدِي سَفَرَةٍ ٭ كِرَامٍ بَرَرَةٍ
﴾
[33]
مُكرمين در فرهنگ قرآن كريم همين ملائكه‌اند ديگر اين شهيد مي‌گويد خداي سبحان ما را به ملائكه ملحق كرده است.
بنابراين ﴿
وَجَعَلَنِي مِنَ الْمُكْرَمِينَ
﴾
[34]
يعني از اينها قرار داد نه «معهم» قرار داد من در رديف ملائكه‌ام جزء ملائكه الآن به حساب مي‌آيم، خب اين مقام به وسيلهٴ همين شهادت حاصل شده است اول كه نبود، پس يك كرامت نفسي است، يك كرامت كسبي آن كرامت كسبي اگر به صورت شهادت باشد انسان را در رديف ملائكه قرار مي‌دهد همان بيان نوراني حضرت امير(سلام الله عليه) كه در
نهج‌البلاغه
بود براي دربار اموي نوشت كه «
تَزْكِيَةِ الْمَرْءِ نَفْسَهُ لَذَكَرَ ذَاكِرٌ فَضَائِلَ جَمَّةً
»
[35]
خودستايي چيز بدي است و دين نهي كرده و اگر نهي شارع مقدس نبود ما مي‌گفتيم كه اهل‌بيت چه فضايلي دارند و ما چه هستيم ولي اجمالش اين است كه خيليها به جبهه مي‌روند شهيد مي‌شوند از خاندان ما كسي رفت شهيد شد مي‌شود سيدالشهداء اين حمزه است كه عموي من است، خيليها به جبهه مي‌روند جانباز مي‌شوند اعضايشان را از دست مي‌دهند از خاندان ما كسي برود جبهه جانباز بشود دستش را بدهد مثل برادرم جعفر ذات اقدس الهي به اين دوتا بال مي‌دهد اين مي‌شود جعفر طيّار در بعضي از نقلها دارد كه «
يطير بهما مع الملائكة في الجنّة
»
[36]
اين مي‌شود جعفر طيّار، اگر در سورهٴ مباركهٴ «فاطر» دارد فرشتگان ﴿
أُوْلِي أَجْنِحَةٍ مَّثْنَي وَثُلاَثَ وَرُبَاعَ
﴾
[37]
اينها داراي جناح‌اند داراي بال‌اند نه پَر آن بالهايشان پر دارد اينها داراي بال‌اند فرمود برادرم جعفر طيّار هم دوتا بال پيدا كرده كه با ملائكه همسفر است و هم‌پَرواست، خب اين با ملائكه بودن يعني با مُكرمين بودن.
فتحصّل كه كرامت يك بار ارزشي دارد كه كبير و عظيم و امثال ذلك فاقد آن هستند و انسان در گوهر ذات خودش اين گوهر گرانمايه را به همراه دارد و اگر به وسيله تقوا يا شهادت يا علل و عوامل ديگر اينها را برجسته‌تر كند، پُر فروغ‌تر كند با ملائكه محشور خواهد شد اينها گوشه‌اي از اوصاف كرامت است و آنچه را كه جناب امام رازي و ديگران ذكر كردند اينها جزء فروعات مسئله است.

اعوذ بالله من الشيطان الرجيم
بسم الله الرحمن الرحيم
﴿
وَلَقَدْ كَرَّمْنَا بَنِي آدَمَ وَحَمَلْنَاهُمْ فِي الْبَرِّ وَالْبَحْرِ و
َرَزَقْنَاهُم مِنَ الطَّيِّبَاتِ وَفَضَّلْنَاهُمْ عَلَي كَثِيرٍ مِمَّنْ خَلَقْنَا تَفْضِيلاً﴿70
﴾
بعد از اينكه بخشي از نعمتهاي الهي را ذكر فرمود و اينكه در حال خطر انسان متوجّه ذات اقدس الهي مي‌شود و خداوند را با چشم دل مشاهده مي‌كند و بعد از شهود از او كمك مي‌طلبد، نه مفهوم «رَبُّما» يا «الهُمّا» را طرح كرده باشد زيرا از مفهوم كاري ساخته نيست تا انسان به او التجا پيدا كند آن‌گاه از سرشت انسان و حقيقت انسان خبر داد فرمود: ﴿
وَلَقَدْ كَرَّمْنَا بَنِي آدَمَ
﴾ و روشن شد كه اين مربوط به حقيقت انسان است اختصاصي به شخص معيّن ندارد اين گوهرْ گرامي و گرانبهاست البته اين حقيقت و اين ذات ممكن است واجب نيست و فقيرِ الي‌الله است و غني نيست لذا كرامت آن هم به تكريم الهي است نه براي خودش، اگر دربارهٴ ذات اقدس الهي آمد كه خداوند كريم است چون هستيِ او عين ذات اوست كرامت او هم ذاتي است ولي اينجا اگر گفته مي‌شود كه اين مربوط به حقيقت انسان است حقيقت انسان و ذات انسان «فقير الي الله» است كه ﴿
يَا أَيُّهَا النَّاسُ أَنتُمُ الْفُقَرَاءُ إِلَي اللَّهِ
﴾
[1]
اين ذاتي كه «فقير الي الله» است و مخلوق خداست خداوند اين ذات را گرامي و نيكو آفريد معناي ذاتي‌بودن كرامت براي انسان همين است يعني عين ذات اوست كه ذات او ممكن است و «فقر الي الله» هست و خداوند هم با تكريم ياد كرد.
مطلب بعدي آن است كه در اين آيه پنج فضيلت براي انسان ذكر شده است كه بعضيها نفسي‌اند، بعضيها نسبي. كرامتْ نفسي است يعني اين حقيقت يك حقيقت گرانمايه است چه اينكه فرشته يك حقيقت گرانمايه است كه از فرشته‌ها به عنوان كِرام ياد مي‌شود ﴿
بِأَيْدِي سَفَرَةٍ ٭ كِرَامٍ بَرَرَةٍ
﴾
[2]
يا ﴿
إِنَّ عَلَيْكُمْ لَحَافِظِينَ ٭ كِرَاماً كَاتِبِينَ
﴾
[3]
و مانند آن. اين حقيقت گرامي است حالا اگر كسي با سوء اختيار خود اين حقيقت را بيراهه مصرف بكند بيراه ببرد او خودش معاقب است وگرنه يك حقيقت گرامي را ذات اقدس الهي آفريد براي اينكه فرمود: ﴿
فَأَلْهَمَهَا فُجُورَهَا وَتَقْوَاهَا
﴾
[4]
او را با الهام خودش ملهم كرده است و فرشته‌ها يا موجودات ديگر دستور الهام نداد خودش الهامِ انسان را به فضيلت و رذيلت به عهده گرفت ﴿
فَأَلْهَمَهَا فُجُورَهَا وَتَقْوَاهَا
﴾، خب پس اين مي‌شود كريم.
اينكه فرمود: ﴿
وَلَقَدْ كَرَّمْنَا بَنِي آدَمَ
﴾ آن كرامتهاي تحصيلي كه در سورهٴ مباركهٴ «حجرات» است آنها به اختيار خود انسان است كه ﴿
إِنَّ أَكْرَمَكُمْ عِندَ اللَّهِ أَتْقَاكُمْ
﴾
[5]
چون اين انسان ذاتاً گرامي داشته شد كمالات بعدي كه خداي سبحان ذكر مي‌كند به استناد همين كرامت اوست. يكي از آن كمالات اين است كه ﴿
وَحَمَلْنَاهُمْ فِي الْبَرِّ
﴾ حَمل انسان در بَرّ يعني ما به انسان توفيقي داديم كه صنايع برّي درست كند نظير اتومبيل و امثال اتومبيل يا حيوانات را براي او ذلول و نَرم و مطيع قرار داديم كه از اين حيوانات استفاده كند مستحبّ است كه انسان وقتي سوار مركب شد به استناد آن آيه معروف بگويد ﴿
سُبْحَانَ الَّذِي سَخَّرَ لَنَا هَذا وَمَا كُنَّا لَهُ مُقْرِنِينَ ٭ وَإِنَّا إِلَي رَبِّنَا لَمُنقَلِبُونَ
﴾
[6]
فرمود ما اين وسيلهٴ نقليه را در اختيار شما قرار داديم حالا يا كشتي درياست يا اسب و امثال اسب در صحراست شما ﴿
لِتَسْتَوُوا عَلَي ظُهُورِهِ ثُمَّ تَذْكُرُوا نِعْمَةَ رَبِّكُمْ إِذَا اسْتَوَيْتُمْ عَلَيْهِ وَتَقُولُوا سُبْحَانَ الَّذِي سَخَّرَ لَنَا هَذا وَمَا كُنَّا لَهُ مُقْرِنِينَ
﴾
[7]
الآن هم مستحب است كسي سوار اتومبيل مي‌شود همين آيه را بخواند ديگر اين اختصاصي به اسب ندارد فرمود اين كار را ما كرديم ﴿
لِتَسْتَوُوا عَلَي ظُهُورِهِ ثُمَّ تَذْكُرُوا نِعْمَةَ رَبِّكُمْ إِذَا اسْتَوَيْتُمْ عَلَيْهِ وَتَقُولُوا سُبْحَانَ الَّذِي سَخَّرَ لَنَا هَذا وَمَا كُنَّا لَهُ مُقْرِنِينَ
﴾.
پس خدا اين وسيلهٴ نقليه را در اختيار انسان قرار داد اگر آن حيوان است او را مطيع انسان قرار داد و اگر وسيله نقليه است توفيق ساخت و سازش را به انسان عطا كرد و اين وسيله شده مسخّر در اختيار انسان و انسان به اذن الهي او را تسخير كرده است يا ساخت و از او بهره‌برداري مي‌كند گرچه از اين آيه ﴿
وَحَمَلْنَاهُمْ فِي الْبَرِّ وَالْبَحْرِ
﴾ غير از كشتي و وسايل سوارشدن مثل اسب و استر و امثال ذلك استفاده نمي‌شود لكن قبلاً در سورهٴ مباركهٴ «نحل» گذشت كه قرآن كريم يك سلسله مخلوقات را هم به نحو اجمال اشاره كرده است آيهٴ هشت سورهٴ مباركهٴ «نحل» اين بود ﴿
وَيَخْلُقُ مَا لاَ تَعْلَمُونَ
﴾ چيزهايي كه الآن شما نمي‌دانيد بعداً خدا براي شما خلق مي‌كند، خب اين وسيله نقليه‌اي كه اسب و استر و حمار باشد اين را كه مي‌دانستند، كشتي را هم كه مي‌دانستند، اين جريان هواپيما و اتومبيل و سفينه‌هاي فضايي و امثال ذلك را نمي‌دانستند فرمود در آينده چيزهايي ديگري را هم خدا براي شما خلق مي‌كند ﴿
وَالْخَيْلَ وَالْبِغَالَ وَالْحَمِيرَ لِتَرْكَبُوهَا وَزِينَةً وَيَخْلُقُ مَا لاَ تَعْلَمُونَ
﴾ بنابراين الآن هم اگر كسي سوار سفينهٴ فضايي شد مستحب است بگويد ﴿
سُبْحَانَ الَّذِي سَخَّرَ لَنَا هَذا

لقد حذفوا أوّل الحديث

لقد حذفوا أوّل الحديث
» جريانش اين است كه اين روايتي كه از پيامبر نقل شد قصّه‌اش اين است «
إنّ رسول الله(صلّي الله عليه و آله و سلّم) مرّ برجلين يتاسابّان
» مرور كرد ديد دو نفر دعوا دارند يكديگر را سبّ مي‌كنند فحش مي‌دهند «
فَسمع أحدهما يقول لصاحبه قبّح الله وجهك ووجه مَن يُشبهك
» حضرت شنيد كه يكي به ديگري مي‌گويد خدا صورتت را زشت كند يا «قَبَّحَ» زشت كرده است و كسي است كه صورت او شبيه صورت توست او را هم زشت كرده است يا نفرين است كه فعل ماضي به معني مضارع باشد در آينده خدا صورت تو را زشت بكند و صورت كسي كه شبيه توست هم زشت بكند «
قَبَّح الله وجهك ووجه من يشبهك
» كه شبيه توست. در چنين صحنه‌اي كه يكديگر را سبّ مي‌كردند وجود مبارك پيغمبر(صلّي الله عليه و آله و سلّم) فرمود: «
يا عبدالله لا تَقُل هذا
» اين طور فحش نده صورت او را و صورت شبيه او را فحش نده «
لا تَقُل هذا لأخيك فإنّ الله عزّ وجلّ خَلق آدم علي صورته
» خداوند آدم را بر صورت او خلق كرد اين ضمير «صورته» به خود آدم برمي‌گردد يعني هر كسي را به صورت مناسب خود برمي‌گرداند يا نه، ضمير «صورته» به «الله» برمي‌گردد خداوند انسان را به صورت خودش خلق كرده است بنابراين صورةالرحمان دارد و صورت رحمان زشت نيست، خب.
اين يك حديث كه خيلي شفّاف و گويا نيست كه ضمير «علي صورته» به خود آن آدم برمي‌گردد يا به «الله» برمي‌گردد و مانند آن. لكن در صفحهٴ 103 روايت هيجده از باب شش، باب شش از ابواب
توحيد
صدوق عنوان باب اين است كه «باب أنّه عزّ وجل ليس بجسمٍ ولا صورةٍ» روايت هيجده اين باب صفحهٴ 103 كتاب صدوق اين است كه محمدبن‌مسلم مي‌گويد «سألتُ أباجعفر(عليهما السلام)» از وجود مبارك امام باقر سؤال كردم «عمّا يروون أنّ الله عزّ وجل خلق آدم علي صورته» من به حضرت عرض كردم اين روايتي كه نقل مي‌كنند خداوند آدم را بر صورت خود خلق كرد اين يعني چه؟ فرمود: «
هي صورةٌ مُحدثةٌ مخلوقةٌ
» صورت آدم قديم و ازلي و اينها نيست حادث است يك، خالق دارد دو، «
اصطفاها الله واختارها علي سائر الصوَر المختلفة
» ذات اقدس الهي صورت انسان را نسبت به ساير صوَر برگزيد، اختيار كرد، انتخاب كرد «
فأضافها الي نفسه
» اين صورت را به خودش اضافه كرد «
كما أضاف الكعبة الي نفسه والروح الي نفسه
» با اينكه او منزّه از خانه و مسكن است به ابراهيم و اسماعيل(سلام الله عليهما) فرمود: ﴿
أَن طَهِّرَا بَيْتِيَ لِلطَّائِفِينَ وَالْعَاكِفِينَ
﴾
[14]
و مانند آن با اينكه او منزّه از جزء و امثال ذلك است روح را به خودش اسناد داد فرمود: ﴿
وَنَفَخْتُ فِيهِ مِن رُّوحِي
﴾
[15]
نه اينكه از روحِ خودم جزء روح خودم ـ معاذ الله ـ به او دادم روحِ انسان مخلوق خداست، كعبه مخلوق خداست بر اساس شرافتي كه براي كعبه است آن را به خود اضافه كرد، اسناد داد فرمود: ﴿
بَيْتِيَ
﴾ بر اساس شرافتي كه براي روح آدمي است آن را به خودش اضافه كرد، اسناد داد فرمود: ﴿
وَنَفَخْتُ فِيهِ مِن رُّوحِي
﴾ صورت انسان هم بشرح ايضاً [همچنين] بر اساس شرافتي كه دارد به خودش اسناد داد لذا پيغمبر(صلّي الله عليه و آله و سلّم) فرمود: «
إنّ الله سبحانه وتعالي خلق آدم علي صورته
»
[16]
همان طوري كه آيات «يفسّر بعضه بعضا»
[17]
احاديث هم همين طور است كه «يفسّر بعضه بعضا» به كمك روايت هيجده باب شش كه در صفحهٴ 103 است روشن مي‌شود كه آن روايتي كه قبلاً خوانده شد ضمير «علي صورته» به «الله» برمي‌گردد نه به «آدم» «إنّ الله خلق آدم» را علي صورت خودش با اينكه خدا منزّه از صورت است همان طوري كه منزّه از خانه است ولي كعبه را بيت‌الله مي‌گوييم منزّه از روح جزيي است در حالي كه روح را به خودش اسناد داد فرمود: ﴿
وَنَفَخْتُ فِيهِ مِن رُّوحِي
﴾
[18]
مُبرّاي از صورت است ولي اين صورت را با اضافهٴ تشريفي به خداي سبحان اسناد داده مي‌شود فرمود: «اصطفاه الله واختارها علي سائر الصوَر المختلفة فأضافها الي نفسه كما أضاف الكعبة إلي نفسه» يك نمونه «والروح الي نفسه» دو نمونه، اين دو نمونه را در اين دو آيه ذكر كرد فقال: ﴿
بَيْتِيَ
﴾
[19]
كه شاهد نمونه اول است، و قال: ﴿
وَنَفَخْتُ فِيهِ مِن رُّوحِي
﴾ كه شاهد نمونهٴ دوم است.
پس بنابراين انسان يك حقيقت گرانمايه است اين حقيقت گرانمايه را اگر هدر داد خودش مسئول است اين حقيقت بي‌سرمايه نيست ذات اقدس الهي با سرمايه اين را خلق كرد بسياري از علوم را به او داد و او را در مشهَد ﴿
أَلَسْتُ بِرَبِّكُمْ قَالُوا بَلَي
﴾
[20]
در آن مشهد اين را حاضر كرده در آن صحنه پيمان گرفته از او خودش را به او نشان داده او نظير اين آينه اگر كسي در برابر آينه قرار بگيرد چون مي‌دانيد در آينه چيزي نيست فقط صورتي كه انسان مي‌بيند در حقيقت براي خود شخص است يعني اگر درختي در برابر آينه باشد ما هم آنجا قرار بگيريم نور چشم ما مي‌خورد به سطح آينه كه جرم شفاف است يك، اين نور برمي‌گردد به درخت دو، ما درخت را در زاويه عطف مي‌بينيم سه، خيال مي‌كنيم در آينه ديديم چهار، وگرنه در آينه چيزي نيست.
مرآتي را كه عُرف مي‌گويد همين شيشه و جيوه و قاب است آينه‌اي كه در بازار سخن از آينه‌فروشي است همين است ولي وقتي در كتاب معرفت مي‌گويند مرآت به اين شيشه و جيوه و قاب نمي‌گويند به آن صورت مي‌گويند مرآت اين شيشه مرآت است يعني وسيله است كه به سبب او ما آن صورت را مي‌بينيم ولي آن صورت وسيله است كه ما به سبب او صاحب صورت را مي‌بينيم كلّ جهان طبق آن روايت نوراني كه از همين
توحيد
مرحوم صدوق قبلاً خوانديم كلّ جهان مثل صوَر مرئيه‌اند در يكي از خطبه‌هاي نوراني حضرت امير هست كه جهان مرائي خداست حالا مرائي را اختلاف كردند كه جمع مرئي است يا جمع مرآت است كلّ عالم صورت مئاتيه است منتها صورت مئاتيه‌اي كه ما عادت كرديم اين بايد شيشه و جيوه و جرم املس و شفّاف باشد حالا يا آينه است يا سنگ است يا آب است يك جرم شفّافي بايد باشد آيا چنين صورتي بدون جرم شفّاف محال است يا نه؟ كه ما دليلي بر استحاله‌اش پيدا نكرديم كلّ اين عالم صوَر مئاتيه حق‌تعالي است همه دارند او را نشان مي‌دهند اگر همه دارند او را نشان مي‌دهند انسان در چنين صحنه‌اي اگر كسي در برابر آينه قرار بگيرد هيچ كس هم نباشد فقط خود اين شخص است و آينه سرِ آينه را خم بكند كه آينه خودش را ببيند اگر سرِ آينه را خم كرد كه آينه خودش را ديد اين صاحب صورت به آينه مي‌گويد كيست؟ مي‌گويد تو. حرفي براي گفتن ندارد كه آينه كه خودش را در آينه نمي‌بيند كه ما چنين حالتي را گذرانديم ﴿
وَإِذْ أَخَذَ رَبُّكَ مِن بَنِي آدَمَ مِن ظُهُورِهِمْ ذُرِّيَّتَهُمْ وَأَشْهَدَهُمْ عَلَي أَنْفُسِهِمْ
﴾ سرِ ما را خم كرد گفت چه كسي را مي‌بينيد؟ ما گفتيم تو را، همين. ﴿
أَلَسْتُ بِرَبِّكُمْ قَالُوا بَلَي
﴾
[21]
ما چنين حقيقتي را به همراه داريم اگر اين را گُم نكنيم البته در همان چهار صفحه رواياتي كه هست اين است كه فرمودند فرشتگان خادمان مؤمنان‌اند مرحوم كليني نقل كرده، ديگران هم نقل كردند در همين چهار صفحه بحث روايي هست كه مؤمن اعزّ از كعبه است، مؤمن بالاتر از كعبه است
[22]
اين گوهرها را ما داريم حالا خود اهل‌بيت(عليهم السلام) كه از حَملهٴ عرش بالاترند آنها روايات فراواني است كه بحث خاصّ خودش را دارد ولي انسان اين حقيقت گرامي را دارد ما در مَشهَدي، در محضري حضور پيدا كرديم ما را آوردند تا خدا خودش را به ما نشان بدهد ما هم ديديم و گفتيم تو ربّي و ما عبديم و اين صحنه هم‌اكنون با ما هست كه خدا دارد احتجاج مي‌كند مي‌فرمايد: ﴿
وَإِذْ
﴾
[23]

اعوذ بالله من الشيطان الرجيم
بسم الله الرحمن الرحيم
﴿
وَلَقَدْ كَرَّمْنَا بَنِي آدَمَ وَحَمَلْنَاهُمْ فِي الْبَرِّ وَالْبَحْرِ و
َرَزَقْنَاهُم مِنَ الطَّيِّبَاتِ وَفَضَّلْنَاهُمْ عَلَي كَثِيرٍ مِمَّنْ خَلَقْنَا تَفْضِيلاً﴿70
﴾
قرآن كريم كه در حقيقت براي هدايت انسان نازل شده است همان طوري كه در بحثهاي قبل ملاحظه فرموديد خود را به عنوان نور معرفي كرد ﴿
قَدْ جَاءَكُم مِنَ اللّهِ نُورٌ وَكِتَابٌ مُبِينٌ
﴾
[1]
پس اوّلين مطلب اين است كه قرآن در معرّفي خود از طرف ذات اقدس الهي خود را نور معرفي كرد.
مطلب دوم آن است كه اوّلين كار نور تفسير قلمرو خودش است وقتي نور مي‌تابد اشيا را روشن مي‌كند ما در تاريكي اشيا را تشخيص نمي‌دهيم وقتي شبِ تار كسي در بيابان قرار گرفت نه كوه را مي‌بيند، نه درخت را، نه چاه را و نه راه را و نه آب را و نه خاك را جايي را تشخيص نمي‌دهد وقتي شمس طالع شد اوّلين كاري كه آفتاب مي‌كند اين است كه اشيا را تفسير مي‌كند كه مشخص مي‌كند چه چيزي آب است، چه چيزي خاك است، چه چيزي درخت است، چه چيزي سنگ است، چه چيزي كوه است، كجا راه است، كجا چاه است بعد از تفسير، تكميل مي‌كند تغيير مي‌دهد ميوه‌ها را مي‌پزاند، درختها را بارور مي‌كند و مانند آن.
قرآن هم اوّلين كاري كه كرده انسان را به انسان معرفي كرده چه اينكه جهان را هم به انسان معرفي كرده كه فرمود آسمان و زمين خلقت‌اند، دريا و صحرا خلقت‌اند ديگر سخن از طبيعت نيست و انسان هم بدني دارد و روحي دارد بدنش از راههاي خاص به اين صورت درآمده است سورهٴ مباركهٴ «مؤمنون» اشاره مي‌كند و روح را هم تا حدودي اينجا كه لازم بود مشخص كرد، گاهي هم انسان را با اجزاي درونيِ او به خود او معرفي مي‌كند كه بدني دارد مضغه و علقه و عظام و لَحم و جنين و گاهي هم از راه روح او را معرفي مي‌كند، گاهي مجموعه را با لحاظ بيروني معرفي مي‌كند كه اين مجموعه نسبت به بيرون چه حُكمي دارد؟
در اين آيه محلّ بحث كه عنوان بني‌آدم مطرح است معلوم مي‌شود آن خصوصيّاتي كه ذات اقدس الهي درباره حضرت آدم مطرح كرد كه او خليفةالله است، مسجود فرشته‌ها هست، از علم اسما باخبر است، توفيق اِنبا و گزارش آن معارف نسبت به فرشته‌ها را دارد همه اين چند امر را ملحوظ كرد فرمود: ﴿
يَا بَنِي آدَمَ
﴾
[2]
آنجا كه خصوصيتهاي آدم مطرح است از انسان به عنوان بني‌آدم ياد مي‌شود گاهي هم ﴿
يَا أَيُّهَا الْإِنسَانُ
﴾
[3]
است، ﴿
يَا أَيُّهَا النَّاسُ
﴾
[4]
است و مانند آن، ولي آ‌نجا كه خصوصيّتهاي حضرت آدم(سلام الله عليه) مطرح است از انسان به عنوان بني‌آدم ياد مي‌كند چه اينكه در جريان عهدگيري ﴿
وَإِذْ أَخَذَ رَبُّكَ مِن بَنِي آدَمَ مِن ظُهُورِهِمْ ذُرِّيَّتَهُمْ
﴾
[5]
اينجا هم عنوان بني‌آدم را مطرح كرده است، اين هم يك مطلب.
مطلب ديگر اينكه در جريان خلقت انسان، معرفي انسان به خود انسان آن بخش بدني او را ذكر كرد كه ﴿
أَلَمْ يَكُ نُطْفَةً مِن مَنيٍّ يُمْنَي
﴾
[6]
تا رساند به عظام و ﴿
فَكَسَوْنَا الْعِظَامَ لَحْماً
﴾ اين بخش، آن بخش ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾
[7]
را به عنوان ﴿
نَفَخْتُ فِيهِ مِن رُوحِي
﴾
[8]
ياد مي‌كند و كرامتهايي كه براي انسان قائل است به لحاظ همان جنبهٴ ﴿
أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾ كه اصلاً اين يك چيز ديگري است و موجودات عالم خلقت يك چيز ديگرند ما او را يك چيز ديگر كرديم يعني از سنخ موجودات مشهود شما نيست ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾ و اين خلقِ آخر با ﴿
نَفَخْتُ فِيهِ مِن رُوحِي
﴾ رابطه دارد اگر آن صبغهٴ ﴿
خَلْقاً آخَرَ
﴾ را حفظ كرد، ﴿
نَفَخْتُ فِيهِ مِن رُوحِي
﴾ را حفظ كرد، ﴿
إِنِّي جَاعِلٌ فِي الأَرْضِ خَلِيفَةً
﴾
[9]
را حفظ كرد مي‌تواند امانت الهي را تحمل بكند آنچه در بخش پاياني سورهٴ مباركهٴ «احزاب» هست كه كاري از انسان ساخته است كه از آسمانها و زمين و جبال ساخته نيست ﴿
إِنَّا عَرَضْنَا الْأَمَانَةَ عَلَي السَّماوَاتِ وَالْأَرْضِ وَالْجِبَالِ فَأَبَيْنَ أَن يَحْمِلْنَهَا وَأَشْفَقْنَ مِنْهَا وَحَمَلَهَا الْإِنسَانُ
﴾
[10]
پس كاري از انسان ساخته است كه از سماوات و ارض ساخته نيست و اگر آ‌ن جنبهٴ ﴿
أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾ را رعايت نكرد، ﴿
إِنِّي جَاعِلٌ فِي الأَرْضِ خَلِيفَةً
﴾ را رعايت نكرد، ﴿
نَفَخْتُ فِيهِ مِن رُوحِي
﴾ را محترم نشمرد به تك‌تك اين موجودات اشاره مي‌كند دربارهٴ بسياري از اينها مي‌گويد همه اينها از انسان بالاتر است مي‌فرمايد: ﴿
لَخَلْقُ السَّماوَاتِ وَالْأَرْضِ أَكْبَرُ مِنْ خَلْقِ النَّاسِ
﴾
[11]
، ﴿
ءَأَنتُمْ أَشَدُّ خَلْقاً أَمِ السَّماءُ بَنَاهَا
﴾
[12]
خب شما اگر با قدرتهاي بدني بخواهيد خودتان را ارزيابي كنيد اين آسمانها كه از شما سنگين‌ترند، زمين كه از شما سنگين‌تر است ﴿
إِنَّكَ لَن تَخْرِقَ الْأَرْضَ وَلَن تَبْلُغَ الْجِبَالَ طُولاً
﴾
[13]
اين كوهها از شما سنگين‌ترند، بلندترند و عظيم‌ترند پس اگر به لحاظ ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾ شد كاري از انسان ساخته است كه از هيچ كدام از سماوات و ارض ساخته نيست برابر بخش پاياني سورهٴ مباركهٴ «احزاب» اگر آن جنبه نبود جنبهٴ بدني بود فرمود زميني كه از رويش راه مي‌روي از شما سنگين‌تر است، كوهي كه در دامنهٴ او به سر مي‌بري از شما بالاتر است، آسماني كه سقف او بر شما سايه انداخت از شما بالاتر است ﴿
ءَأَنتُمْ أَشَدُّ خَلْقاً أَمِ السَّماءُ بَنَاهَا
﴾، ﴿
لَخَلْقُ السَّماوَاتِ وَالْأَرْضِ أَكْبَرُ مِنْ خَلْقِ النَّاسِ
﴾، ﴿
إِنَّكَ لَن تَخْرِقَ الْأَرْضَ وَلَن تَبْلُغَ الْجِبَالَ طُولاً
﴾ بعد نسبت به حيوانات هم مي‌فرمايد: ﴿
أُولئِكَ كَالْأَنْعَامِ بَلْ هُمْ أَضَلُّ
﴾
[14]
، گاهي مي‌فرمايد: ﴿
فَهِيَ كَالْحِجَارَةِ أَوْ أَشَدُّ قَسْوَةً
﴾
[15]
پس اگر جنبهٴ بدني باشد و آن حرمت خلافت و كرامت الهي رعايت نشود تك‌تك اين موجودات را مي‌شمارد مي‌گويد از شما بهترند، بزرگ‌ترند، زخيم‌ترند، عظيم‌‌ترند و مانند آن، اگر آن جنبهٴ ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾ رعايت بشود مي‌شود ﴿
فَضَّلْنَاهُمْ عَلَي كَثِيرٍ مِمَّنْ خَلَقْنَا
﴾ بلكه در بين انسان كساني‌اند كه نسبت به همه مخلوقات برترند براي اينكه آن امانتهاي الهي را بدون ظلم و جهل حمل مي‌كنند، خلافت الهي را بدون نقص حمل مي‌كنند، تعلّم اسماي حسناي الهي را بدون جهل فرامي‌گيرند، گزارش آنها نسبت به ملائكه را بدون سهو به عهده دارد يا ﴿
أَنْبِئُونِي بِأَسْمَاءِ هؤُلاَءِ
﴾
[16]
هم مي‌شوند معلّم ملائكه، هم مي‌شوند متعلِّم ذات اقدس الهي و هم مي‌شوند خليفهٴ خدا، خليفهٴ خدا در همان بحثهاي اوايل سورهٴ مباركهٴ «بقره» ملاحظه فرموديد كه آن خلافتي كه در كلام و حكمت مطرح است در حدّ نبوت مطرح است كه انسان چون به اصطلاح مدني بالطبع است زندگي‌اش زندگي جمعي است و زندگي جمعي بدون قانون نمي‌شود قانون را بايد كسي وضع كند كه به همه اين مسائل احاطهٴ علمي داشته باشد يك و از غرض‌ورزي هم منزّه باشد دو، مبتلا به جهل و سهو و نسيان هم نشود سه، چنين كسي جز ذات اقدس الهي كس ديگري نيست و آن هم بايد به وسيله پيامبر پيام خود را به مردم برساند اين نياز بشر به پيامبر است اين همان برهان معروفي است كه در كتابهاي حكمت و كلام هست و در روايات ما هم قسمت مهم بر اساس همين جهت تكيه شده يعني كتاب شريف
اصولي كافي
بخش اضطرار به حجّت كه كتاب امامت است اوّلين حديث و دوّمين حديث همين است كه وقتي از وجود مبارك امام رضا، امام صادق(سلام الله عليهما) يا ساير ائمه سؤال مي‌كنند به چه دليل ما نيازمند به حجّتيم مي‌فرمايند به همين دليل كه انسان بالأخره بدون قانون نمي‌شود قانون‌گذار بايد عالِم باشد و منزّه از سهو و نسيان و جهل باشد همين برهان معروف را كه در حقيقت اول انبيا و اوليا ارائه كردند بعد كه حكما فهميدند همين برهان معروف كلامي هست اما وقتي كه سخن از خلافت است در فرهنگ قرآن انسان خليفةالله هست نه تنها خليفهٴ خداست كه جامعه را اداره كند بلكه خليفهٴ خداست هم اصلاح جامعه را به عهده بگيرد، هم تعليم ملائكه را به عهده بگيرد مثل اينكه آنها هم يك آموزش و پرورشي دارند مسئول آموزش و پرورش فرشته‌ها هم بايد انسان باشد و هم بهشت‌سازي را به عهده مي‌گيرند آنچه كه از قرآن برمي‌آيد اضلاع سه‌گانهٴ اين مثلث است كه ما چرا به انسان كامل يا به عنوان وليّ، وصيّ، نبيّ، خليفةالله نيازمنديم؟ براي سه جهت يكي اينكه اين ﴿
يُعَلِّمُهُمُ الْكِتَابَ وَالْحِكْمَةَ
﴾
[17]
كه تأمين علميِ جامعه به عهدهٴ انبياست دربارهٴ فرشته‌ها هم هست فرشته‌ها هم معلّم مي‌خواهند آن فصل خصومت ملأ اعلا هم به عهدهٴ انبياي الهي است نه تنها فصل خصومت در دستگاه قضايي به عهدهٴ انبياست بلكه حلّ خصومت ملأ اعلا هم به عهدهٴ انبياست.
پس ما نيازمند به نبيّ هستيم هم براي رفع نيازهاي اجتماعي بشر، هم براي رفع نيازهاي علمي فرشته‌ها، هم براي تأمين ساخت و ساز بهشت، خب بالأخره بهشت را چه كسي بايد بسازد؟ بايد بشر بسازد ديگر مهندسي‌اش چيست؟ معماري‌اش چيست؟ مصالح ساختماني‌اش چيست؟ راه‌حلّش چيست؟ چطوري ما بهشت را بسازيم؟ چگونه ما غُرف مَبنيهٴ ابدي بسازيم كه هرگز ويران نشود؟ چگونه درختي غرس كنيم كه ابدالدهر از بين نرود و هر وقت خواستيم ميوه بدهد؟ اينها بايد به ما بگويند وگرنه بهشت طبق بيان نوراني خود ائمه(عليهم السلام) كه فرمودند: «
الجنّة قيعان
»
[18]
زمين بهشت صاف است چيزي در آن نسيت اين شما هستيد كه داريد براي آن بهشت خانه مي‌سازيد و درخت مي‌كاريد و چشمه مي‌جوشانيد كه ﴿
يُفَجِّرُونَهَا تَفْجِيراً
﴾
[19]
، خب.
پس آنچه كه در كتابهاي حكمت و كلام است يك سوم اين مجموعهٴ عظيم است اين دائرةالمعارفي است كه قابل قياس نيست با آنچه كه در كتابهاي حكمت و كلام است آنها مي‌گويند بشر پيغمبر مي‌خواهد براي اينكه مدني‌ بالطبع است و قانون مي‌خواهد اما قرآن مي‌گويد بشر پيغمبر مي‌خواهد براي اينكه وجود پيغمبر لازم است براي اينكه هم بشر را تأمين كند، هم ملائكه را تأمين بكند هم به مردم ياد بدهد چطوري در دنيا زندگي كنيد و چطوري بهشت را بسازيد اين كارها به وسيلهٴ خليفةالله هست و اين كار فقط از انسان ساخته است قهراً از ضميمه‌كردن آيه محلّ بحث كه دارد ﴿
وَفَضَّلْنَاهُمْ عَلَي كَثِيرٍ مِمَّنْ خَلَقْنَا
﴾ كه اين به تعبير سيدناالاستاد ناظر به موجودات عالم كُوْن است يعني نسبت به گياهان و حيوانات و موجودات عالم مشهود و محسوس مادي و جِرمي و هم به شهادت بخش پاياني سورهٴ مباركهٴ «احزاب» انسان نسبت به كلّ سماوات و ارض افضل است و هم نسبت به ملاحظهٴ بخشهاي اوايل سورهٴ مباركهٴ «بقره» كه ﴿
إِنِّي جَاعِلٌ فِي الأَرْضِ خَلِيفَةً
﴾
[20]
بعد ﴿
وَعَلَّمَ آدَمَ الْأَسْماءَ
﴾
[21]
بعد ﴿
يَا آدَمُ أَنْبِئْهُمْ بِأَسْمَائِهِمْ
﴾
[22]
كه آن مي‌شود معلّم ملائكه قهراً انسان نه فردِ انسان در انسان حقيقتي هست، افرادي هستند كه افضل علي الجميع‌اند بالقول المطلق و آن وجود مبارك پيغمبر(صلّي الله عليه و آله و سلّم) است، اهل‌بيت است، انبياي الهي‌اند، مرسلين‌اند، اولواالعزم‌اند اين‌گونه از بزرگان اينهايي كه معلم ملائكه‌اند، معماران بهشت‌اند و سازندگان عالم طبيعت هستند.
فتحصّل كه از خود اين آيه به تنهايي ما بخواهيم استفاده كنيم كه انسان از جميع ما سوي‌الله بر جميع ما سوي‌الله رُجحان دارد و افضل از همه است كار آساني نيست اما وقتي تحليل بكنيم بگوييم كه اين آيه ناظر به همان موجودات حيواني و موجودات مكتوم و امثال ذلك است كه فرمود: ﴿
مَا مِن دَابَّةٍ فِي الْأَرْضِ وَلاَ طَائِرٌ يَطِيرُ بِجَنَاحَيْهِ إِلَّا أُمَمٌ أَمْثَالُكُمْ
﴾
[23]
آنها هم مثل شما هستند اُممي هستند مثل شما آن وقت اين آيه مي‌خواهد بفرمايد انسان از همه آنها بالاتر است ولي به ضميمهٴ بخش پاياني سورهٴ مباركهٴ «احزاب» انسان از مجموعهٴ سماوات و ارض بالاتر است به ملاحظهٴ بخش اوايل سورهٴ مباركهٴ «بقره» انسان از همه ملائكه افضل است، بنابراين اگر رواياتي دارد كه انبيا(عليهم السلام)، اهل‌بيت(عليهم السلام) افضل از ملائكه هستند اين از مجموعه اين آيات به خوبي برمي‌آيد.
مطلب ديگر اينكه ذات اقدس الهي در سورهٴ مباركهٴ «سجده»، «فصلت» و مانند آن اين بخش را در قرآن ما زياد مي‌بينيم كه فرمود: ﴿
وَنَفَخْتُ فِيهِ مِن رُّوحِي
﴾
[24]
، ﴿
فَإِذَا سَوَّيْتُهُ وَنَفَخْتُ فِيهِ مِن رُّوحِي
﴾ وقتي من اين را تسويه كردم، بدن او را مستوي‌الخِلقه كردم و ساختار بدني او را تأمين كردم اعضاي دروني و بيروني او را ساختم نه به صورت يك سراميك دستي بلكه دستگاه گوارش به او دادم، مغز دادم، قلب دادم همه اعضا و ساختار دروني و بيروني او را تأمين كردم ﴿
فَإِذَا سَوَّيْتُهُ وَنَفَخْتُ فِيهِ مِن رُّوحِي فَقَعُوا لَهُ سَاجِدِينَ
﴾ روح را در او دَميدم و به او افاضه كردم براي او سجده كنيد كه اين مي‌شود مسجودٌله شما، خب.
اين با آن رواياتي كه دارد «
خَلق الله الأرواح قبل الأجساد
»
[25]
با آنها هماهنگ است معلوم مي‌شود روح انسان قبلاً بود و ذات اقدس الهي روح را كه جداي از انسان است به انسان متعلّق كرد. اما در سورهٴ مباركهٴ «مؤمنون» كه تطوّرات اين خلقت را ذكر مي‌كند ظاهرش اين است كه روح تكامل‌يافتهٴ همين نشئهٴ طبيعي بدن است آيه يازده به بعد سورهٴ مباركهٴ «مؤمنون» اين است ﴿
وَلَقَدْ خَلَقْنَا الْإِنسَانَ مِن سُلاَلَةٍ مِن طِينٍ
﴾ اين براي اصل انسان، نسلش از نطفه است ﴿
ثُمَّ جَعَلْنَاهُ نُطْفَةً فِي قَرَارٍ مَّكِينٍ ٭ ثُمَّ خَلَقْنَا النُّطْفَةَ عَلَقَةً فَخَلَقْنَا الْعَلَقَةَ مُضْغَةً فَخَلَقْنَا الْمُضْغَةَ عِظَاماً فَكَسَوْنَا الْعِظَامَ لَحْماً ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾ يعني همان را ما چيز ديگر كرديم اين با حركت جوهري و جسمانيةالحدوث و روحانيةالبقاء هماهنگ است نه اينكه چيز ديگر از خارج آورديم همان را چيز ديگر كرديم نه اينكه «أعطيناه شيئاً آخر» اين ضمير به همان مخلوق قبلي برمي‌گردد ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾
[26]
او را چيز ديگر كرديم نه اينكه چيز ديگر به او داديم.
بنابراين آن آيه با روايات «
خَلق الله الأرواح
»
[27]
هماهنگ است، اين آيه با جسمانيةالحدوث و روحانيةالبقا هماهنگ است كه بايد در مسئله معرفت نفس اينها جمع‌بندي بشود چه اينكه اين كار را كردند آن روايات را حمل كردند بر نشئهٴ عقلي انسان كه قبل از بدن بود، اين را حمل كردند بر نشئهٴ لفظي خب، بنابراين همين را ما متحوّل كرديم به صورت چيز ديگر درآورديم و همين مي‌شود خليفهٴ خدا.
مطلب ديگر در جريان خلافت است كه ارزش انسان به همان خليفةالله بودن است. در جريان خلافت ملاحظه فرموديد آن رواياتي كه بخش‌اش در
نهج‌البلاغه
وجود مبارك حضرت امير هست بخش‌اش هم در جاي ديگر يك خلافت متقابلي بين خدا و بندهٴ اوست. بخش رسمي خلافت اين است كه انسان خليفةالله هست ﴿
إِنِّي جَاعِلٌ فِي الأَرْضِ خَلِيفَةً
﴾
[28]
اين معنا دارد محتاج به توضيح است كه چگونه انسان خليفةالله مي‌شود و دشوار هم هست. بخش ديگر كه در
نهج‌البلاغه
و در بخشهاي روايي ائمه(عليهم السلام) است اين است كه خدا خليفهٴ انسان است خدا خليفهٴ انسان باشد از بعضي از جهات قابل درك است يعني انسان در خيلي از موارد حضور ندارد كسي بايد باشد در غياب او كار او را انجام بدهد اين دعاي سفري كه در
نهج‌البلاغه
هست همين است كه مستحب است مسافر هنگام سفر اين دعا را بخواند «
اللَّهُمَّ أَنْتَ الصَّاحِبُ فِي السَّفَرِ وَ أَنْتَ الْخَلِيفَةُ فِي الْأَهْلِ وَ لاَ يَجْمَعُهُما غَيْرُكَ لِأَنَّ الْمُسْتَخْلَفَ لا يَكُونُ مُسْتَصْحَباً وَ الْمُسْتَصْحَبَ لاَ يَكُونُ مُسْتَخْلَفاً
»
[29]
انسان مستحبّ است وقتي خواست سفر كند بگويد خدايا همراه من باش مرا در سفر حفظ كن، در غياب من نزد بچه‌هاي من خليفهٴ من باش كه اينها را اينجا حفظ كني و اين كار فقط از تو ساخته است براي اينكه آنكه نزد بچه‌هاي ماست با ما نيست، آنكه با ماست نزد بچه‌هاي ما نيست «
وَ لاَ يَجْمَعُهُما غَيْرُكَ لِأَنَّ الْمُسْتَخْلَفَ لا يَكُونُ مُسْتَصْحَباً وَ الْمُسْتَصْحَبَ لاَ يَكُونُ مُسْتَخْلَفاً
».
اين معنا حالا يا تعارف هست يا تعبيرات ديگر بالأخره قابل درك هست براي اينكه انسان در بعضي از موارد غيبت دارد و چون حضور ندارد كسي بايد باشد در غياب كار او را انجام بدهد اما درباره ذات اقدس الهي ترسيم خلافتش بسيار دشوار است براي اينكه نه ذات اقدس الهي غيبت دارد و نه خِفايي دارد و نه در جايي عجزي دارد كجاست كه او نيست تا انسان بتواند كار او را انجام بدهد او اگر غيبتي ندارد، اگر او خفايي ندارد، اگر او تعطيلي در كار او نيست در همه جا هست ديگر خلافت معنايش چيست؟
توضيحش مي‌تواند اين امر باشد او كه ﴿
بِكُلِّ شَيْ‏ءٍ مُحِيطٌ
﴾
[30]
است اگر او ﴿
بِكُلِّ شَيْ‏ءٍ مُحِيطٌ
﴾ است خلافت هم نظير نبوّت، نظير رسالت مقوله به تشكيك است درجاتي دارد دربارهٴ نبوّت و رسالت دو آيه بحثش گذشت كه ﴿
تِلْكَ الرُّسُلُ فَضَّلْنَا بَعْضَهُمْ عَلَي بَعْضٍ
﴾
[31]
يك، ﴿
لَقَدْ فَضَّلْنَا بَعْضَ النَّبِيِّينَ عَلَي بَعْضٍ
﴾
[32]
دو، انبيا با هم متفاوت‌اند، مرسلين با هم متفاوت‌اند اين يك تفاوت درون‌گروهي است بر خلاف فضيلت انسان نسبت به غير انسان كه يك فضيلت برون‌گروهي است يك وقت مي‌گوييم اين نوع نسبت به انواع ديگر افضل‌اند، يك وقت مي‌گوييم در بين خود انبيا بعضهم نسبت به بعض افضل‌اند. اگر نبوّت درجاتي دارد، رسالت درجاتي دارد خلافت هم بشرح ايضاً [همچنين] بعضيها خليفه‌اند در يك محدودهٴ خاصّي خليفةالرازق‌اند، خليفةالشافي‌اند، خليفةالقابض‌اند، خليفةالباسط‌اند بعضيها خليفةالمحيط‌اند اگر خليفةالمحيط شدند اگر ذات اقدس الهي همه‌ جا حضور دارد اين شخصي كه خليفهٴ محيط است هم بايد همه جا حضور داشته باشد هم بتواند در ملأ اعلا فصل خصومت كند، هم در جوامع بشري فصل خصومت كند، هم در بهشت بهشتيها را باشد هم در صحنهٴ قيامت آنجا خلافت الهي را إعمال بكند هم در ملأ اعلا و اين در انسان كاملي مثل وجود مبارك پيغمبر(صلّي الله عليه و آله و سلّم) قابل تصوّر است خليفهٴ خدايي كه ﴿
بِكُلِّ شَيْ‏ءٍ مُحِيطٌ
﴾
[33]
است و «مع كل شيء» هست با همه اشيا هست لا بالممازجه چنين انسان كاملي كه همه جا حضور داشته باشد با انبيا باشد، با مرسلين باشد، با احيا باشد، با اموات باشد، با برزخيها باشد، با اهل دنيا باشد، با اهل آخرت باشد چنين انسان كاملي مي‌شود خليفهٴ مطلق، وگرنه خليفهٴ مطلق نخواهد بود آنكه صادر اول است يا ظاهر اول است او مي‌تواند خليفهٴ خدا باشد بالقول المطلق. فتحصّل كه انسان در بخش جسمش حُكم خاصّي دارد كه در سورهٴ مباركهٴ «مؤمنون» بيان كرديم و در بخش روحش حُكم خاصّي دارد كه بخشي از اينها در همان سوره آمده، بخشي هم در سورهٴ ديگر.
مطلب دوم اينكه آياتي كه مربوط به روح است بخشي از اينها با خلقت روح قبل از بدن هماهنگ است، بخشي از اينها با تحوّل جوهري و جسمانيةالحدوث و روحانيةالبقاء بودن و مانند آن مناسبت‌تر است كه دارد ما همان را چيز ديگر كرديم نه اينكه به انسان چيز ديگر داديم ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾
[34]
نه «أعطيناه خلقاً آخر».
مطلب سوم هم اين شد كه اگر انسان بر اساس همين جنبهٴ حيواني‌اش فكر بكند خداي سبحان مي‌فرمايد كه همه آنچه كه مي‌بيني از تو بزرگ‌تر و شريف‌تر و امثال ذلك، بر اساس جنبهٴ الهي‌اش و روحاني‌اش و مجرّد نفسش فكر بكند كاري از او ساخته است كه از سماوات و ارض ساخته نيست در بين همين انسانها كساني‌اند كه نه تنها كاري از او ساخته است كه از سماوات و ارض ساخته نيست از ملائكه هم ساخته نيست نه تنها از ملائكه ساخته نيست بلكه ملائكه بايد از او فرابگيرند كه ﴿
يَا آدَمُ أَنْبِئْهُمْ بِأَسْمَائِهِمْ
﴾
[35]
.
بنابراين اگر رواياتي وارد شده است كه ملائكه افضل نسبت به انسان هستند اين في‌الجمله درست است نه بالجمله، اگر گفته شد انسانها افضل از ملائكه هستند اين هم في‌الجمله درست است نه بالجمله براي اينكه همه انسانها كه از ملائكه بالاتر نيستند بعضي از انسانها همانهايي هستند كه فرمود: ﴿
أُولئِكَ كَالْأَنْعَامِ بَلْ هُمْ أَضَلُّ
﴾
[36]
است يا ﴿
شَيَاطِينَ الْإِنْسِ وَالْجِنِّ
﴾
[37]
است و مانند آن اينها را ما بايد هر كدام را در مقطع و موطن خاصّ خودشان ارزيابي بكنيم و نگوييم انسان بالقول المطلق افضل است و اينها و دربارهٴ هيچ مخلوقي هم ذات اقدس الهي به عنوان ﴿
فَتَ
بَارَكَ اللَّهُ أَحْسَنُ الْخَالِقِينَ
﴾
[38]
نفرمود مگر درباره انسان، اينكه دربارهٴ انسان فرمود: ﴿
فَتَبَارَكَ اللَّهُ أَحْسَنُ الْخَالِقِينَ
﴾ كه در سورهٴ مباركهٴ «مؤمنون» آمده اين به مناسبت آن ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾ است وگرنه آن بخش اول كه ﴿
ثُمَّ جَعَلْنَاهُ نُطْفَةً فِي قَرَارٍ مَّكِينٍ ٭ ثُمَّ خَلَقْنَا النُّطْفَةَ عَلَقَةً فَخَلَقْنَا الْعَلَقَةَ مُضْغَةً فَخَلَقْنَا الْمُضْغَةَ عِظَاماً فَكَسَوْنَا الْعِظَامَ لَحْماً
﴾
[39]
تا اينجا كه در گوسفند و غير گوسفند هم يكي است آنها هم همين را مي‌گويند اين ديگر ﴿
فَتَبَارَكَ اللَّهُ أَحْسَنُ الْخَالِقِينَ
﴾ ندارد كه، اما از اينكه فرمود: ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾ اين «فاء» تفريع بر اين جملهٴ اخير است نه بر آن قبيلها ﴿
فَتَبَارَكَ اللَّهُ أَحْسَنُ الْخَالِقِينَ
﴾ پس اگر انسان احسن‌المخلوقين نباشد خداي او هم احسن‌الخالقين نخواهد بود اگر ذات اقدس الهي احسن‌الخالقين مي‌شود براي اينكه انسان را چيز ديگر كرده است آن قدر اين حقيقت پيچيده است كه نفرمود ما او را چه چيزي كرديم فرمود يك چيز ديگر كرديم آن همان است كه اگر كسي خودش را بشناسد خداي خود را يقيناً مي‌شناسد و ما در اضطرار و امثال ذلك مفهوم را درك نمي‌كنيم چون مفهوم را الآن هم درك مي‌كنيم در حال عادي هم درك مي‌كنيم التجا هم نداريم كسي به مفهوم دل نمي‌بندد مفهوم «ربّما»، مفهوم «الٰهمّا» به مصداق گُنگ هم متوسّل نمي‌شويم در آن حالت غرق و احساس خطر ما يك حقيقت شفّاف و روشني را اول مي‌بينيم بعد به او سر مي‌ستاييم در آستان او سر مي‌ساييم از او كمك مي‌خواهيم به او اميدواريم اينكه ﴿
دَعَوُا اللَّهَ مُخْلِصِينَ لَهُ الدِّينَ
﴾
[40]
شد يعني «عرفوه معرفة ايقانٍ» اولاً، بعد ﴿
دَعَوُا اللَّهَ مُخْلِصِينَ لَهُ الدِّينَ
﴾ ثانياً نه اينكه به مفهوم «الهمّا» مفهوم به درد آن انسان غريق نمي‌خورد او به مفهوم اميدوار نيست مفهوم در حال عادي هم هست.

أَلَا يَعْلَمُ مَنْ خَلَقَ
﴾

وَنَبْلُوكُم بِالشَّرِّ وَالْخَيْرِ فِتْنَةً
﴾
[42]

أَنزَلَ لَكُم مِنَ الْأَنْعَامِ ثَمَانِيَةَ أَزْوَاجٍ

أَنزَلَ لَكُم مِنَ الْأَنْعَامِ ثَمَانِيَةَ أَزْوَاجٍ
﴾ يا ﴿
أَنزَلْنَا الْحَدِيدَ فِيهِ بَأْسٌ شَدِيدٌ
﴾

اينكه در آنجا بود كه همين آهن سنگين و وَزين نبود كه اينكه در سورهٴ مباركهٴ «حجر» بحثش گذشت ﴿
إِن مِّن شَيْ‏ءٍ إِلَّا عِنْدَنَا خَزَائِنُهُ وَمَا نُنَزِّلُهُ إِلَّا بِقَدَرٍ مَعْلُومٍ
﴾
[45]
هر چيزي خزائني دارد نه يك مخزن، نه يك خزينه «إنْ مِن شيء إلاّ عندنا خزينته» نيست، «مخزنه» نيست ﴿
إِلَّا عِنْدَنَا خَزَائِنُهُ
﴾ آن مِن شيء است يعني «لكلّ شيء خزائن» نه مجموع براي مجموع است «لكل شيء خزائنه» چندين مرحله هست از هر مرحله عبور مي‌كند تا به اينجا برسد آهن از آنجا آمده، اين حيوانات از آنجا آمدند آهن از آنجا آمده برابر سورهٴ مباركهٴ «حديد» اين حيوانات از آنجا آمده برابر سورهٴ مباركهٴ «زمر» ﴿
أَنزَلْنَا
﴾ به معني «خَلقنا» نيست ﴿
أَنزَلْنَا
﴾ به معني ﴿
أَنزَلْنَا
﴾ است معلوم مي‌شود آنجا وجود داشتند مناسب با آنجا احكامي داشتند حِكَمي داشتند بعد وقتي تنزّل كردند ... معصومين هم همين طور هستند آنجا كه هستند با وجود نوري موجودند كه ذات اقدس الهي مي‌‌داند كه اينها اگر وارد نشئهٴ طبيعت بشوند به هر امري امتحان بشوند پيروزمندانه از امتحان سربرمي‌دارند، خب.

﴾ احسن‌الخالقين، احسن‌المخلوقين دارد، احسن‌الصوَركم دارد، احسن تقويم دارد و مانند آن، خب.

پاسخ: بله، همان اينكه گفته شد دشوار است براي اينكه غيبتي ندارد. در جريان خلافت الهي انسان محتاج است كه خليفهٴ خدا باشد نه خدا محتاج است كه خدا خليفهٴ او باشد در همين قُرب نوافل كه اين را هم مرحوم كليني نقل كرد هم در كتب اهل سنّت آمده است انسان به جايي مي‌رسد كه ذات اقدس الهي زبان او مي‌شود، چشم او مي‌شود، گوش او مي‌شود و در قرب فرايض گرچه اين رواياتش در
كافي
و اينها نيست انسان به جايي مي‌رسد كه زبانِ بي‌زباني خدا مي‌شود درباره وجود مبارك حضرت امير هست كه «
عين الله
» هست، «
يد الله
» هست. بخشي از اين روايات را مرحوم صدوق(رضوان الله عليه) در كتاب شريف توحيد صدوق نقل كرده از وجود مبارك حضرت امير(سلام الله عليه) كه «
أنا عين الله
»، «
أنا جَنب الله
»
[48]
و مانند آن. در قُرب فرايض انسان چشمِ بي‌چشمي خدا مي‌شود در قُرب نوافل ذات اقدس الهي زبانِ انسان مي‌شود ولي چه قُرب فرايض، چه قرب نوافل همه اين بحثها در فصل سوم است چون آن دو فصل يعني مقام هويّت مطلقهٴ ذات و مقام صفات ذات كه عين ذات است بالقول المطلق منطقهٴ ممنوعه است احدي به آنجا راه ندارد در همين وجه‌الله است، فيض‌الله است، ﴿
اللَّهُ نُورُ السَّماوَاتِ وَالْأَرْضِ
﴾
[49]
است كه مقام ظهور خدا، فيض خدا، عنايت خداست كه همين داراي كرامت و شرافت و جلال است، همين كه داراي كرامت و جلال است انسان را كريم كرده است اگر دارد ﴿
لَقَدْ كَرَّمْنَا بَنِي آدَمَ
﴾ ما انسان را مُكرّم كرديم به او كرامت داديم براي آن است كه وجه خدا كريم است در جريان كرامت وجه الهي ذات اقدس الهي هم وجه خدا را كريم مي‌داند، هم خود مقام ربّ را مقام كريم مي‌داند در سورهٴ مباركهٴ «الرحمن» ملاحظه مي‌فرماييد آيه سورهٴ مباركهٴ «الرحمن» اين است كه ﴿
كُلُّ مَنْ عَلَيْهَا فَانٍ ٭ وَيَبْقَي وَجْهُ رَبِّكَ ذُو الْجَلاَلِ وَالْإِكْرَامِ
﴾
[50]
نه «ذِي‌الجلال» نه اينكه ربّ ذي‌الجلال است، وجه ذي‌الجلال است و اين صحيح نيست كه بگوييم از سِنخ نعت مقطوع است و «هو» در تقدير است و «هو» را به ربّ برگردانيم بگوييم «هو» يعني ربّ ذو‌الجلال است، خير خود اين «ذو» صفت است و متعلّق مي‌شود مرتبط مي‌شود به وجه چه وجه خدا داراي جلال اكرام است همين وجه جليل و با جلال همين وجه كريم و با اكرام انسان را كريم كرده است البته اين كرامتها به مقام بالاتر هم وابسته است كه در همين سورهٴ مباركهٴ «الرحمن» در بخش پاياني اين‌‌چنين آمده است يعني آيهٴ 78 سورهٴ «الرحمن» اين است كه ﴿
تَبَارَكَ اسْمُ رَبِّكَ
﴾ كه اينجا مجرور است ﴿
تَبَارَكَ اسْمُ رَبِّكَ
﴾ كه اين ربّ ﴿
ذِي الْجَلاَلِ وَالْإِكْرَامِ
﴾ است، خب آنجا كه فرمود: ﴿
وَيَبْقَي وَجْهُ رَبِّكَ
﴾
[51]
آن وجه داراي جلالت و اكرام است همان وجه‌الله انسان را كريم كرده است اگر كسي وجه‌الله شد «وجيها عند الله» و من المقرّبين شد مي‌شود كريم اگر دربارهٴ وجود مبارك امام مجتبي گفته مي‌شود كريم اهل‌بيت است يا درباره وجود مبارك امام رضا(سلام الله عليه) گفته مي‌شود كريم است يا دربارهٴ اين بي‌بي(سلام الله عليها) گفته مي‌شود كريمهٴ اهل‌بيت است براي اينكه اينها با وجيه عندالله‌اند، وجه‌الله‌اند و من المقرّبين‌اند اگر كسي وجه‌الله شد، وجيه عندالله شد چون وجه‌الله، ذوالجلال والاكرام است اينها هم جليل‌اند، هم كريم اگر شدند و مثل آنكه درباره حضرت عيسي(سلام الله عليه) آمده ﴿
وَجِيهاً فِي الْدُّنْيَا وَالْآخِرَةِ وَمِنَ الْمُقَرَّبِينَ
﴾
[52]
كه اينها وقتي وجيه شدند وجه‌الله با جلالت همراه است، وجه‌الله با اكرام همراه است اينها مي‌شوند كريم آن وقت كرامت اينها در فضايل علمي و عملي اينها تعبيه شده است و منشأ پيدايش كرامتهاي ديگر هم خواهد بود.

اعوذ بالله من الشيطان الرجيم
بسم الله الرحمن الرحيم
﴿
يَوْمَ نَدْعُوا كُلَّ أُنَاسٍ بِإِمَامِهِمْ فَمَنْ أُوتِيَ كِتَابَهُ بِيَمِينِهِ فَأُولئِكَ يَقْرَءُونَ كِتَابَهُمْ وَلاَ يُظْلَمُونَ فَتِيلاً
﴿71﴾
وَمَن كَانَ فِي هذِهِ أَعْمَي فَهُوَ فِي الْآخِرَةِ أَعْمَي وَأَضَلُّ سَبِيلاً ﴿7
2
﴾
چون سورهٴ مباركهٴ «اسراء» در مكه نازل شد و عناصر محوري سُوَر مكّي اصول اعتقادي و همچنين خطوط كلي اخلاق و فقه و حقوق است نه جزئيات و فرعيّات فقهي و حقوقي و مانند آن و جريان معاد هم از اصول كلّي اعتقاد هست لذا همان طوري كه در بخشي از آيات قبل راجع به معاد مطالبي فرمودند الآن هم درباره معاد مطلبي را ارائه مي‌دهند.
مطلب دوم آن است كه خلقت انسان در دنيا با خلقت انسان در آخرت فرق مي‌كند در دنيا انسان اصلي دارد و نسلي هم اصلش با وضع خاص پديد آمد، هم نسلش با وضع مخصوص پديد مي‌آيد. در جريان اصل انسان فرمود اول تراب بود، بعد طين شد، بعد ﴿
حَمَإٍ مَّسْنُونٍ
﴾
[1]
شد، بعد ﴿
صَلْصَالٍ كَالْفَخَّارِ
﴾
[2]
شد، بعد ﴿
نَفَخْتُ فِيهِ مِن رُوحِي
﴾
[3]
و مانند آن.
درباره نسل هم فرمود نطفه بود عَلقه شد، مضغه شد، استخوان شد استخوان را گوشت پوشانديم به صورت جنين شد ﴿
ثُمَّ أَنشَأْنَاهُ خَلْقاً آخَرَ
﴾
[4]
آن دربارهٴ اصل انسان، اين هم دربارهٴ نسل انسان ولي درباره معاد نه اصلش نظير اصل دنياست، نه نسلي نظير نسل دنيا دارد آنجا اصلاً سخن از نسل نيست كه با توالد و توليد افراد به دنيا بيايند همه دفعتاً از خاك سر برمي‌دارند لذا مسئلهٴ نَسَب و مسئله نژاد و قبيله و خانواده در قيامت اصلاً مطرح نيست براي اينكه همه دفعتاً از خاك زنده مي‌شوند و اگر دربارهٴ آيات معاد آمده است كه پدر از پسر فرار مي‌كند يا برادر از برادر فرار مي‌كند اين به علاقهٴ «ما كان» است يعني كسي كه در دنيا برادر ديگري بود امروز از او فرار مي‌كند، كسي كه در دنيا پدر بود امروز از پسر فرار مي‌كند و بالعكس نه اينكه در آنجا پدري و پسري مطرح است.
پرسش:...﴿
إِلَي أَهْلِهِ مَسْرُوراً
﴾ يعني چه؟
پاسخ: يعني به لحاظ كساني كه در دنيا اهلشان بودند ﴿
يَنقَلِبُ إِلَي أَهْلِهِ مَسْرُوراً
﴾
[5]
وگرنه در آخرت نسلي در كار نيست.
بنابراين اصل پيدايش افراد در معاد به سبك پيدايش افراد در دنيا نيست.
مطلب سوم آن است كه اينها با دعوت الهي ذات اقدس الهي همين كه بخواند، دعوت كند اينها زنده مي‌شوند دعوت هم دو قِسم است يك دعوت تشريعي است كه فرمود خدا افراد را به هدايت دعوت مي‌كند يا به پيغمبر(صلّي الله عليه و آله و سلّم) فرمود: ﴿
ادْعُ إِلَي سَبِيلِ رَبِّكَ بِالْحِكْمَةِ وَالْمَوْعِظَةِ الْحَسَنَةِ
﴾
[6]
و مانند آن. اينها دعوتهاي تعليمي، تدريسي، دعوتهاي تشريعي است اينها با اختيار مخاطب همراه است گاهي اطاعت است، گاهي عصيان است و مانند آن.
پرسش:...
پاسخ: بله، حالا اين بعد از ورود در بهشت و جهنم است اما الآن در طليعهٴ معاد كه افراد مي‌خواهند زنده بشوند محور بحث است، خب.
وقتي مي‌خواهند افراد اين دعوتي كه در معاد هست دعوت تكويني است نه دعوت تشريعي و اعتباري دعوت تشريعي اين است كسي را هدايت مي‌كنند، راهنمايي مي‌كنند اختيار اين شخص فاصله است مي‌خواهد بپذيرد، مي‌خواهد نپذيرد خداوند هدايت مي‌كند، دعوت مي‌كند انبيا دعوت مي‌كنند بعضي مي‌پذيرند بعضي نمي‌پذيرند چون دعوت تشريعي است اما دعوت تكويني ديگر عصيان‌پذير نيست ذات اقدس الهي وقتي كسي را دعوت كرد او حتماً بايد اجابت بكند و اجابت هم مي‌كند اينكه فرمود: ﴿
يَوْمَ نَدْعُوا
﴾ اين دعوت تكويني است نه دعوت تشريعي لذا عصيان‌پذير نيست همگان سر از قبر برمي‌دارند اين هم يك مطلب.
مطلب ديگر اين است كه دعوت به چه نام است يك، دعوت به چه سوي است اين دو، حالا مراحل بعدي را قرآن ذكر مي‌كند اما در اين مرحله كه دعوت به چه نام است و دعوت به كدام سَمت است اينها را در همين بخش از آيات سورهٴ مباركهٴ «اسراء» مي‌شود فهميد. در اينجا فرمود: ﴿
يَوْمَ نَدْعُوا كُلَّ أُنَاسٍ بِإِمَامِهِمْ
﴾ گرچه ممكن است ذات اقدس الهي دفعتاً هر كسي را به نام مخصوص او صدا بزند كه زيد است، عمرو است، بَكر است، خالد است، وليد است هر چه هست هيچ محذوري ندارد كه دفعتاً واحده افراد را به اسامي آنها صدا بزند و آنها زنده بشوند لكن در اين كريمه فرمود ما افراد را به امامشان دعوت مي‌كنيم مي‌گوييم علوي بيايد، حَسني بيايد، حسيني بيايد، آنهايي كه اين راه را رفته‌اند بيايند اما اگر عده‌اي جزء ائمه كفر بودند ﴿
وَجَعَلْنَاهُمْ أَئِمَّةً يَدْعُونَ إِلَي النَّارِ
﴾
[7]
به اسم آنها صدا مي‌زنيم آنها كه كافر و منافق بودند در آن رديف صف مي‌بندند پس ﴿
يَوْمَ نَدْعُوا كُلَّ أُنَاسٍ بِإِمَامِهِمْ
﴾.
مطلب بعدي آن است كه جناب زمخشري در
كشّاف
دارد كه اين «مِن بِدَع التفسير» از مطالب بديع و نكات تازه تفسيري اين است كه اينها مي‌گويند اين امام جمع اُمّ است ﴿
يَوْمَ نَدْعُوا كُلَّ أُنَاسٍ بِإِمَامِهِمْ

نكته ديگر براي اظهار شرف امام حسن و امام حسين(سلام الله عليهما) است كه اينها را به احترام فاطمه(سلام الله عليها) افراد را به اسم مادرشان مي‌خوانند، اين دو نكته.

سيدناالاستاد(رضوان الله عليه) مايلند كه اجتماع وجودي دارد و كتابي دارد و احكام و حِكَمي دارد. اشكالي كه بر اين حرف هست اين است كه وحدت مساوق با وجود است اگر چيزي واحد نباشد وجود نيست، وحدت نداشته باشد وجود ندارد جامعه وحدت ندارد وحدتش اعتباري است قهراً وجود ندارد وجودش اعتباري است كه ان‌شاءالله پاسخش بايد در ذيل آيه سورهٴ مباركهٴ «جاثيه» مطرح بشود.
مطلب ديگر كه آن هم مهم است البته بحثهاي آن اهمّ از اين مطلب است منتها بحثهاي اول عميق و علمي است بر اساس آن كتابها نوشته شده كه آيا جامعه وجود دارد يا وجود ندارد و اينها. اما اين مطلب دوم كه اهمّ از مطلب اول است چون يك مطلب ناآشنايي است و آن مطلب ناآشنا اين است كه چون مربوط به جريان معاد است و محلّ ابتلاي خيليها نيست نظير وجود جامعه كتابي نوشته بشود بحثي شده باشد اين نيست آن مطلب اين است كه ما يك نامهٴ معهود داريم كه ﴿
وَإِنَّ عَلَيْكُمْ لَحَافِظِينَ ٭ كِرَاماً كَاتِبِينَ ٭ يَعْلَمُونَ مَا تَفْعَلُونَ
﴾
[14]
يا ﴿
رُسُلُنَا لَدَيْهِمْ يَكْتُبُونَ
﴾
[15]
﴿
أَحَاطَ بِمَا لَدَيْهِمْ
﴾
[16]
و مانند آن هر كاري كه ما مي‌كنيم اينها مي‌نويسند اين درست است اين دركش هم آسان است سخت نيست و ظاهر بخشي از ادلّه هم همين است اما اين آيه سورهٴ مباركهٴ «جاثيه» مي‌فرمايد: ﴿
إِنَّا كُنَّا نَسْتَنسِخُ مَا كُنتُمْ تَعْمَلُونَ
﴾
[17]
ما استنساخ مي‌كنيم نه كتابت، كتابت اين كسي كه دارد حرف مي‌زند يا كار انجام مي‌دهد كسي همين حرفها را يا كارهاي او را مي‌نويسد اين آسان است اما استنساخ يعني ما يك نسخهٴ اصل داريم كه بر اساس اين نسخهٴ اصل آن نسخهٴ فرع را تنظيم مي‌كنند از روي اين نسخهٴ اصل پس يك نسخهٴ اصل لازم است، يك نسخهٴ فرع كه اين نسخه‌نويسي را مي‌گويند استنساخ.
مطلب بعدي آن است كه اگر اعمال ما نسخهٴ اصل باشد آنها كه مي‌نويسند مي‌شود گفت استنساخ است يعني رفتار ما و گفتار ما بشود نسخهٴ اصل آنها يعني فرشتگان از روي اين نُسَخ اصلي مطالبي را يادداشت مي‌كنند اين هم قابل تطبيق است كه بپذيريم اين استنساخ است اما آيه مي‌فرمايد آنچه را كه شما عمل مي‌كنيد ما استنساخ مي‌كنيم يك روايت نوراني در ذيل اين آيه است كه وجود مبارك امام(سلام الله عليه) فرمود: «
أوَ لستمُ عُرُبا
»
[18]
مگر شما عرب نيستيد فصاحت عربي نمي‌دانيد و قوانين و ادبيات عرب را نمي‌دانيد خدا مي‌فرمايد: ﴿
مَا كُنتُمْ تَعْمَلُونَ
﴾ را ما استنساخ مي‌كنيم نه نَسخ، نه نسخه‌نويسي مي‌كنيم، نه آنچه را كه شما انجام مي‌دهيد ما برابر او مي‌نويسيم خير، آنچه را كه شما انجام مي‌دهيد اين استنساخ ماست ﴿
نَسْتَنسِخُ مَا كُنتُمْ تَعْمَلُونَ
﴾
[19]
نه «عمّا كنتم تعملون»، نه بر اساس كار شما، كار شما را ما استنساخ مي‌كنيم پس يك نسخهٴ اصلي لازم است كه اين كاري كه شما انجام مي‌دهيد ما يكي پس از ديگري بيرون مي‌كشيم و اين مطابق با آن نسخهٴ اصل است اين جريان قضا و قدر را ثابت مي‌كند، جريان علم پيشين خدا را ثابت مي‌كند منتها با حفظ مبادي و اختيار كه خداي سبحان مي‌داند زيد با مِيل و ارادهٴ و اختيار خود به طرف صلاح مي‌رود با اينكه مي‌تواند به طرف فساد برود هر چه هم او را به طرف فساد دعوت مي‌كنند او با ميل و اختيار خودش به طرف صلاح گرايش دارد ديگري مي‌تواند به طرف صلاح برود و هر چه او را دعوت مي‌كنند به طرف صلاح ولي او به سوء اختيار خود به طرف فساد مي‌رود ما آنچه در نهان و درونِ اوست استنساخ مي‌كنيم يعني يك نسخهٴ اصلي داريم كه نزد ماست آنچه را كه او انجام مي‌دهد مطابق با آن نسخهٴ اصلي انجام مي‌گيرد ﴿
إِنَّا كُنَّا نَسْتَنسِخُ
﴾ نه «نكتب» ﴿
نَسْتَنسِخُ
﴾ نه «عمّا كنتم تعملون» ﴿
نَسْتَنسِخُ مَا كُنتُمْ تَعْمَلُونَ
﴾ اين ﴿
مَا كُنتُمْ تَعْمَلُونَ
﴾ استنساخ ماست پس يك نسخهٴ اصلي نزد خداست اين استدلال وجود مبارك حضرت است در ذيل اين روايت تقريباً مي‌دانم نود درصد اين مطلب حل نشده است شايد ده درصد هم حل نشده باشد اما خب چون البته مبادي‌اش حاصل نيست، رواياتش حاصل نيست اينكه در خود روايات مطالبي است غُرر آنهاست و از اين نكات الآن دهها مفسّر شما مي‌بينيد همه همين‌طور معنا مي‌كنند مي‌گويند يعني ما اعمال شما را به وسيله فرشته‌ها مي‌نويسيم اما وقتي به روايات اهل‌بيت مراجعه مي‌كنيد مي‌گويد مگر ما عرب نيستيد خدا كه نگفت «إنّا نكتبُ»، «إنّا نَكتب» آيات ديگر است ﴿
نَحْنُ نُحْيِ الْمَوْتَي وَنَكْتُبُ مَا قَدَّمُوا وَآثَارَهُمْ
﴾
[20]
يك، ﴿
إِنَّ عَلَيْكُمْ لَحَافِظِينَ ٭ كِرَاماً كَاتِبِينَ
﴾
[21]
اين درست است، ﴿
رُسُلُنَا لَدَيْهِمْ يَكْتُبُونَ
﴾
[22]
اين درست است، اين سه‌ طايفه آيه يك معناي قابل فهم عمومي دارد اما اين از آن قبيل نيست كه كسي يك مختصر درس بخواند بفهمد فرمود ما آنچه در درون شماست استنساخ مي‌كنيم از يك نسخهٴ اصلي ما قبلاً مي‌دانيم شما چه كار هستيد.

پاسخ: آن ديگر علمِ ازلي خداست اينها ان‌شاءالله به لطف الهي وقتي به ذيل آيه سورهٴ «يس» رسيديم حل مي‌شود منتها اين روايت يك مقدار آدم را آرام مي‌كند معلوم مي‌شود كه قواعد عربي، آشنايي به عربي، عربيِ قويِ غني در فهم آيات ضروري و لازم است فرمود: «
أو لستم عُربا
» مگر ادبيات عرب را نمي‌دانيد قرآن به لسان عربيِ مبين آمده اينجا كه نگفته «انا نكتب»، نگفت «ننسخُ»، نگفت «نستنسخ عمّا كنتم تعملون» هيچ كدام از اين تعبيرات را به كار نبرده فرمود عين آنچه را كه شما عمل مي‌كنيد ما استنساخ مي‌كنيم همين نسخه اصل نزد ماست ﴿
إِنَّا كُنَّا نَسْتَنسِخُ
﴾.
به هر تقدير اين مطلب دوم قوي‌تر از آن مطلب اول است كه جامعه وجود دارد، جامعه كتاب دارد، احكام سياسي، اجتماعي افراد را در آن نامهٴ جامعه مي‌نويسند چون دارد ﴿
كُلَّ أُمَّةٍ جَاثِيَةً كُلُّ أُمَّةٍ تُدْعَي إِلَي كِتَابِهَا
﴾
[23]
نه «كلّ فردٍ يُدعيٰ إلي كتابه» آن آيات ديگر است در آيات ديگر دارد هر كسي به نامهٴ عملش دعوت مي‌شود.
بنابراين اگر ما دو دعوت داشتيم و دو كتاب اينها معارض هم نيستند اگر بعضي از ادلّه دلالت مي‌كرد كه هر كسي را دعوت مي‌كنند اين محشور مي‌شود اين ممكن است و اگر آيه محلّ بحث سورهٴ «اسراء» كه دارد ﴿
يَوْمَ نَدْعُوا كُلَّ أُنَاسٍ
﴾ نه «كلّ فرد» ﴿
كُلَّ أُنَاسٍ بِإِمَامِهِمْ
﴾ اين هم يك دعوت است پس ما دو دعوت داريم يك دعوت فردي و يك دعوت جمعي حالا كه دعوت فرديمان مشخص، دعوت جمعي‌مان مشخص شد سؤال مي‌كنيم ما را براي چه كاري احضار كردي؟ يعني «إلي كتابه و إلي كتابها» هر كسي را به نامهٴ عمل او در بخشهاي مسائل فردي، هر امّتي را «الي كتابها» در مسائل سياسي و اجتماعي ﴿
تُدْعَي إِلَي كِتَابِهَا
﴾
[24]
آنجا كه افراد را به كتاب شخصي خودشان دعوت مي‌كنند بخشي از آنها در همين آيه سورهٴ مباركهٴ «اسراء» آمده بخشي هم در سورهٴ مباركهٴ «حاقّه» و آن سَمتها اشاره شده كه فرمود آنهايي كه به كتابشان دعوت مي‌شوند اگر اهل يُمن و بركت باشند جزء اصحاب ميمنت‌اند و به طرف يُمن دعوت مي‌شوند و اگر اهل مشئمه و مشئوم‌بودن و زشتي باشند به طرف مشئمه دعوت مي‌شوند در سورهٴ مباركهٴ «حاقّه» آيهٴ 24 به بعد اين است آيهٴ نوزده اين است ﴿
فَأَمَّا مَنْ أُوتِيَ كِتَابَهُ بِيَمينِهِ فَيَقُولُ هَاؤُمُ اقْرَؤُا كِتَابِيَهْ
﴾ با يك صداي بلند مي‌گويد آقايان بيايد نامهٴ مرا بخوانيد ﴿
إِنِّي ظَنَنتُ أَنِّي مُلاَقٍ حِسَابِيَهْ ٭ فَهُوَ فِي عِيشَةٍ رَاضِيَةٍ ٭ فِي جَنَّةٍ عَالِيَةٍ ٭ قُطُوفُهَا دَانِيَةٌ ٭ كُلُوا وَاشْرَبُوا هَنِيئاً بِمَا أَسْلَفُتُمْ فِي الْأَيَّامِ الْخَالِيَةِ
﴾ اما ﴿
مَنْ أُوتِيَ كِتَابَهُ بِشِمالِهِ فَيَقُولُ
﴾ آن هم مي‌خواند منتها ديگر با نگراني اينكه مي‌گويد ﴿
مَالِ هذَا الْكِتَابِ لاَ يُغَادِرُ صَغِيرَةً وَلاَ كَبِيرَةً إِلَّا أَحْصَاهَا
﴾
[25]
همين است ديگر آ‌‌ن‌گاه ﴿
فَيَقُولُ يَا لَيْتَنِي لَمْ أُوتَ كِتَابِيَهْ ٭ وَلَمْ أَدْرِ مَا حِسَابِيَهْ ٭ يَا لَيْتَهَا كَانَتِ الْقَاضِيَةَ ٭ مَا أَغْنَي عَنِّي مَالِيَهْ ٭ هَلَكَ عَنِّي سُلْطَانِيَهْ
﴾ آنجا خدا فرمود: ﴿
كُلُوا وَاشْرَبُوا هَنِيئاً بِمَا أَسْلَفُتُمْ فِي الْأَيَّامِ الْخَالِيَةِ
﴾ اينجا مي‌فرمايد: ﴿
خُذُوهُ فَغُلُّوهُ ٭ ثُمَّ الْجَحِيمَ صَلُّوهُ
﴾، خب.
پس افراد شده چهار مطلب يكي اينكه به چه نام دعوت مي‌شوند اين دو طايفه است، يك طايفه بالصراحه دارد كه به امام، يكي هم به نحو عموم دارد هر كسي را ما دعوت مي‌كنيم هم به شخصه دعوت مي‌شوند هم به اسم رهبرشان. دو مطلب ديگر اين است كه اگر به شخصه دعوت شدند وقتي كه دعوت شد و زنده شد مي‌گويد مرا براي چه چيزي حاضر كرديد؟ مي‌گويد «إلي كتابك» دعوت به نامهٴ عملت مهمانسراي او همان نامهٴ عمل اوست آنهايي كه به نام جمعيت و اجتماع و رهبر و امام دعوت شدند مي‌گويد كجا بايد برويم؟ مي‌گويند به كتاب عمومي، به كتاب سياسي و اجتماعي‌تان كه ﴿
كُلُّ أُمَّةٍ تُدْعَي إِلَي كِتَابِهَا
﴾
[26]
.

پاسخ: حالا نه، آن نسخهٴ اصل چيست بايد بماند ولي متن آنچه كه از جان افراد درمي‌آيد اين نسخهٴ دوم است. آنچه كه نزد خداي سبحان است نسخهٴ اصل است كه اين را برابر با آن انجام مي‌دهند آن وقت فرشتگان ديگري هستند كه آدم وقتي اين كارها را انجام داد مي‌نويسند دو گروه‌اند بعضيها كارهاي فردي را مي‌نويسند، بعضيها كارهاي جمعي را مي‌نويسند اينها كارها تقسيم‌شده است ولي يك عده هستند كه آنچه كه در درون افراد نهادينه شده است اين را درمي‌آورند اين را مي‌گويند استنساخ يك نسخهٴ اصل است نزد آنها آنچه كه در درون اينهاست استنساخ مي‌كنند وقتي كه ظهور پيدا كرد آن وقت كرام كاتبين بخشي مربوط به اعمال فردي است را مي‌نويسند، بخشي مربوط به اعمال جمعي است را مي‌نويسند آيا آن بخشهاي سياسي، اجتماعي كه مربوط به كارهاي جمعي است فصلي از فصول اين كتاب است؟ «يُحتمل» يك كتاب جداگانه است؟ «يُحتمل» اين بايد جداگانه بحث بشود ولي غرض اين است كه كارها جداست، مأموريتها هم جدا. در آيه محلّ بحث سورهٴ مباركهٴ «اسراء» به هر دو قسمت اشاره كرده يكي را شفّاف، يكي را به صورت نيمه‌شفّاف آنكه به صورت شفّاف بيان كرده فرمود: ﴿
